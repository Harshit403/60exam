    99→function InlineLineChart({ data, height=200 }: { data:{day:string;signups:number}[]; height?:number }) {
   100→  if (!data.length) return null
   101→  const maxVal = Math.max(...data.map(d=>d.signups), 1)
   102→  const w = 500, h = height, padL = 40, padR = 20, padT = 20, padB = 40
   103→  const plotW = w - padL - padR, plotH = h - padT - padB
   104→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   105→  const xScale = (i:number) => padL + (i/(data.length-1))*plotW
   106→
   107→  const linePath = data.map((d,i)=>`${i===0?'M':'L'}${xScale(i)},${yScale(d.signups)}`).join(' ')
   108→  const areaPath = linePath + ` L${xScale(data.length-1)},${padT+plotH} L${padL},${padT+plotH} Z`
   109→  const yTicks = 5
   110→  const yStep = maxVal / yTicks
   111→
   112→  return (
   113→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   114→      {/* Grid lines */}
   115→      {Array.from({length:yTicks+1}).map((_,i)=>{
   116→        const val = Math.round(i*yStep)
   117→        const y = yScale(val)
   118→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   119→      })}
   120→      {/* Area fill */}
   121→      <path d={areaPath} fill="#10b981" fillOpacity={0.08} />
   122→      {/* Line */}
   123→      <path d={linePath} fill="none" stroke="#10b981" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
   124→      {/* Dots + labels */}
   125→      {data.map((d,i)=><g key={i}>
   126→        <circle cx={xScale(i)} cy={yScale(d.signups)} r="4" fill="#10b981" />
   127→        <text x={xScale(i)} y={h-8} textAnchor="middle" className="fill-muted-foreground" fontSize="11">{d.day}</text>
   128→      </g>)}
   129→    </svg>
   130→  )
   131→}

   133→function InlineBarChart({ data, height=220 }: { data:TopPerformer[]; height?:number }) {
   134→  if (!data.length) return null
   135→  const maxVal = Math.max(...data.map(d=>d.studyHours), 1)
   136→  const barW = Math.min(40, 400/data.length - 8)
   137→  const w = 500, h = height, padL = 40, padR = 20, padT = 10, padB = 60
   138→  const plotW = w - padL - padR, plotH = h - padT - padB
   139→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   140→  const gap = plotW / data.length
   141→
   142→  return (
   143→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   144→      {/* Grid */}
   145→      {[0,0.25,0.5,0.75,1].map((pct,i)=>{
   146→        const y = yScale(maxVal*pct)
   147→        const val = Math.round(maxVal*pct)
   148→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   149→      })}
   150→      {/* Bars */}
   151→      {data.map((d,i)=>{
   152→        const x = padL + i*gap + gap/2 - barW/2
   153→        const barH = (d.studyHours/maxVal)*plotH
   154→        const y = padT + plotH - barH
   155→        const name = d.name.length > 8 ? d.name.slice(0,7)+'…' : d.name
   156→        return <g key={i}>
   157→          <rect x={x} y={y} width={barW} height={barH} fill="#f59e0b" rx="3" className="transition-opacity hover:opacity-80" />
   158→          <text x={padL+i*gap+gap/2} y={h-28} textAnchor="middle" className="fill-muted-foreground" fontSize="10" transform={`rotate(-25,${padL+i*gap+gap/2},${h-28})`}>{name}</text>
   159→        </g>
   160→      })}
   161→    </svg>
   162→  )
   163→}

   165→// ─── Hooks ───────────────────────────────────────────────────────
   166→function useFetch<T>(fetcher:()=>Promise<T>, deps:any[]) {
   167→  const [data, setData] = useState<T|null>(null)
   168→  const [loading, setLoading] = useState(true)
   169→  const [error, setError] = useState<string|null>(null)
   170→  const [refreshKey, setRefreshKey] = useState(0)
   171→  const refresh = useCallback(()=>setRefreshKey(k=>k+1), [])
   172→  useEffect(()=>{
   173→    let c=false
   174→    fetcher().then(r=>{if(c)return;setData(r);setError(null);setLoading(false)}).catch(e=>{if(c)return;setError(e.message||'Failed');setLoading(false)})
   175→    return()=>{c=true}
   176→  }, [...deps, refreshKey])
   177→  return { data, loading, error, refresh, setData }
   178→}

   180→function usePagination<T>(items:T[], pageSize=10) {
   181→  const [page, setPage] = useState(1)
   182→  const totalPages = Math.max(1, Math.ceil(items.length/pageSize))
   183→  const safePage = Math.min(page, totalPages)
   184→  const paginated = items.slice((safePage-1)*pageSize, safePage*pageSize)
   185→  const goTo = (p:number)=>setPage(Math.max(1,Math.min(p,totalPages)))
   186→  if (page>totalPages&&totalPages>0) setPage(totalPages)
   187→  return { page:safePage, totalPages, paginated, goTo, setPage }
   188→}

   192→  const blob = new Blob([csv],{type:'text/csv;charset=utf-8;'})
   193→  const url = URL.createObjectURL(blob)
   194→  const a = document.createElement('a'); a.href=url; a.download=`${filename}.csv`; a.click()
   195→  URL.revokeObjectURL(url)
   196→}

   921→// ─── 10. Change Password Dialog ──────────────────────────────────
   922→function ChangePasswordDialog({ open, onOpenChange }: { open:boolean; onOpenChange:(open:boolean)=>void }) {
   923→  const [currentPassword,setCurrentPassword]=useState(''), [newPassword,setNewPassword]=useState(''), [confirmPassword,setConfirmPassword]=useState(''), [submitting,setSubmitting]=useState(false)
   924→
   925→  const handleSubmit=async(e:React.FormEvent)=>{
   926→    e.preventDefault()
   927→    if(newPassword!==confirmPassword){toast.error('New passwords do not match');return}
   928→    if(newPassword.length<6){toast.error('Password must be at least 6 characters');return}
   929→    setSubmitting(true)
   930→    try{await api.adminChangeOwnPassword(currentPassword,newPassword);toast.success('Password changed successfully');onOpenChange(false);setCurrentPassword('');setNewPassword('');setConfirmPassword('')}
   931→    catch(e:any){toast.error(e.message||'Failed to change password')}finally{setSubmitting(false)}
   932→  }
   933→
   934→  return <Dialog open={open} onOpenChange={onOpenChange}><DialogContent>
   935→    <DialogHeader><DialogTitle>Change Password</DialogTitle><DialogDescription>Update your admin account password</DialogDescription></DialogHeader>
   936→    <form onSubmit={handleSubmit} className="space-y-4">
   937→      <IconField icon={Lock} label="Current Password" type="password" value={currentPassword} onChange={e=>setCurrentPassword(e.target.value)} required />
   938→      <IconField icon={Lock} label="New Password" type="password" value={newPassword} onChange={e=>setNewPassword(e.target.value)} required minLength={6} />
   939→      <IconField icon={Lock} label="Confirm New Password" type="password" value={confirmPassword} onChange={e=>setConfirmPassword(e.target.value)} required minLength={6} />
   940→      <DialogFooter><Button type="submit" disabled={submitting}>{submitting&&<Loader2 className="size-4 animate-spin" />} Change Password</Button></DialogFooter>
   941→    </form>
   942→  </DialogContent></Dialog>
   943→}
