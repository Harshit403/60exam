   352→// ─── 2. Students Page ────────────────────────────────────────────
   353→function StudentsPage() {
   354→  const [students, setStudents] = useState<Student[]>([])
   355→  const [loading, setLoading] = useState(true)
   356→  const [search, setSearch] = useState('')
   357→  const [addOpen, setAddOpen] = useState(false)
   358→  const [editStudent, setEditStudent] = useState<Student | null>(null)
   359→  const [deleteStudent, setDeleteStudent] = useState<Student | null>(null)
   360→  const [passwordStudent, setPasswordStudent] = useState<Student | null>(null)
   361→  const [courses, setCourses] = useState<Course[]>([])
   362→
   363→  const fetchStudents = useCallback(() => {
   364→    setLoading(true)
   365→    api.adminStudents(search || undefined)
   366→      .then((res) => setStudents(res.data || res.students || res))
   367→      .catch((err) => toast.error(err.message))
   368→      .finally(() => setLoading(false))
   369→  }, [search])
   370→
   371→  useEffect(() => { fetchStudents() }, [fetchStudents])
   372→  useEffect(() => {
   373→    api.adminCourses()
   374→      .then((res) => setCourses(res.data || res.courses || res))
   375→      .catch(() => {})
   376→  }, [])
   377→
   378→  const handleAdd = async (e: React.FormEvent<HTMLFormElement>) => {
   379→    e.preventDefault()
   380→    const fd = new FormData(e.currentTarget)
   381→    try {
   382→      await api.adminCreateStudent({
   383→        name: fd.get('name'), email: fd.get('email'), mobile: fd.get('mobile'),
   384→        password: fd.get('password'), courseId: fd.get('courseId'),
   385→      })
   386→      toast.success('Student created')
   387→      setAddOpen(false)
   388→      fetchStudents()
   389→    } catch (err: any) { toast.error(err.message) }
   390→  }
   391→
   392→  const handleEdit = async (e: React.FormEvent<HTMLFormElement>) => {
   393→    e.preventDefault()
   394→    if (!editStudent) return
   395→    const fd = new FormData(e.currentTarget)
   396→    try {
   397→      await api.adminUpdateStudent(editStudent.id, {
   398→        name: fd.get('name'), email: fd.get('email'), mobile: fd.get('mobile'),
   399→        courseId: fd.get('courseId'), status: fd.get('status'),
   400→      })
   401→      toast.success('Student updated')
   402→      setEditStudent(null)
   403→      fetchStudents()
   404→    } catch (err: any) { toast.error(err.message) }
   405→  }
   406→
   407→  const handleDelete = async () => {
   408→    if (!deleteStudent) return
   409→    try {
   410→      await api.adminDeleteStudent(deleteStudent.id)
   411→      toast.success('Student deleted')
   412→      setDeleteStudent(null)
   413→      fetchStudents()
   414→    } catch (err: any) { toast.error(err.message) }
   415→  }
   416→
   417→  const handleChangePassword = async (e: React.FormEvent<HTMLFormElement>) => {
   418→    e.preventDefault()
   419→    if (!passwordStudent) return
   420→    const fd = new FormData(e.currentTarget)
   421→    try {
   422→      await api.adminChangePassword(passwordStudent.id, fd.get('password') as string)
   423→      toast.success('Password changed')
   424→      setPasswordStudent(null)
   425→    } catch (err: any) { toast.error(err.message) }
   426→  }
   427→
   428→  return (
   429→    <div className="space-y-4">
   430→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   431→        <div className="relative w-full sm:w-72">
   432→          <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   433→          <Input placeholder="Search students..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-8" />
   434→        </div>
   435→        <Button onClick={() => setAddOpen(true)}><Plus className="size-4" /> Add Student</Button>
   436→      </div>
   437→
   438→      <Card>
   439→        <CardContent className="p-0">
   440→          {loading ? <div className="p-4"><TableSkeleton cols={6} /></div> : (
   441→            <Table>
   442→              <TableHeader>
   443→                <TableRow>
   444→                  <TableHead>Name</TableHead>
   445→                  <TableHead>Email</TableHead>
   446→                  <TableHead className="hidden md:table-cell">Mobile</TableHead>
   447→                  <TableHead>Course</TableHead>
   448→                  <TableHead>Status</TableHead>
   449→                  <TableHead className="text-right">Actions</TableHead>
   450→                </TableRow>
   451→              </TableHeader>
   452→              <TableBody>
   453→                {!students.length ? (
   454→                  <TableRow><TableCell colSpan={6} className="text-center py-8 text-muted-foreground">No students found</TableCell></TableRow>
   455→                ) : students.map((s) => (
   456→                  <TableRow key={s.id}>
   457→                    <TableCell className="font-medium">{s.name}</TableCell>
   458→                    <TableCell className="text-sm">{s.email}</TableCell>
   459→                    <TableCell className="hidden md:table-cell">{s.mobile}</TableCell>
   460→                    <TableCell>{s.course?.title || '—'}</TableCell>
   461→                    <TableCell>
   462→                      <Badge variant={s.status === 'APPROVED' ? 'default' : s.status === 'REJECTED' ? 'destructive' : 'secondary'}>
   463→                        {s.status}
   464→                      </Badge>
   465→                    </TableCell>
   466→                    <TableCell className="text-right">
   467→                      <div className="flex items-center justify-end gap-1">
   468→                        <Button variant="ghost" size="icon" onClick={() => setEditStudent(s)} title="Edit"><Pencil className="size-4" /></Button>
   469→                        <Button variant="ghost" size="icon" onClick={() => setPasswordStudent(s)} title="Change Password"><Lock className="size-4" /></Button>
   470→                        <Button variant="ghost" size="icon" onClick={() => setDeleteStudent(s)} title="Delete"><Trash2 className="size-4 text-destructive" /></Button>
   471→                      </div>
   472→                    </TableCell>
   473→                  </TableRow>
   474→                ))}
   475→              </TableBody>
   476→            </Table>
   477→          )}
   478→        </CardContent>
   479→      </Card>
   480→
   481→      {/* Add Student Dialog */}
   482→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   483→        <DialogContent>
   484→          <DialogHeader><DialogTitle>Add Student</DialogTitle><DialogDescription>Create a new student account</DialogDescription></DialogHeader>
   485→          <form onSubmit={handleAdd} className="space-y-3">
   486→            <div><Label>Name</Label><Input name="name" required /></div>
   487→            <div><Label>Email</Label><Input name="email" type="email" required /></div>
   488→            <div><Label>Mobile</Label><Input name="mobile" required /></div>
   489→            <div><Label>Password</Label><Input name="password" type="password" required /></div>
   490→            <div>
   491→              <Label>Course</Label>
   492→              <Select name="courseId">
   493→                <SelectTrigger className="w-full"><SelectValue placeholder="Select course" /></SelectTrigger>
   494→                <SelectContent>
   495→                  {courses.map((c) => <SelectItem key={c.id} value={c.id}>{c.title}</SelectItem>)}
   496→                </SelectContent>
   497→              </Select>
   498→            </div>
   499→            <DialogFooter><Button type="submit">Create Student</Button></DialogFooter>
   500→          </form>
   501→        </DialogContent>
   502→      </Dialog>
   503→
   504→      {/* Edit Student Dialog */}
   505→      <Dialog open={!!editStudent} onOpenChange={(open) => !open && setEditStudent(null)}>
   506→        <DialogContent>
   507→          <DialogHeader><DialogTitle>Edit Student</DialogTitle></DialogHeader>
   508→          {editStudent && (
   509→            <form onSubmit={handleEdit} className="space-y-3">
   510→              <div><Label>Name</Label><Input name="name" defaultValue={editStudent.name} required /></div>
   511→              <div><Label>Email</Label><Input name="email" type="email" defaultValue={editStudent.email} required /></div>
   512→              <div><Label>Mobile</Label><Input name="mobile" defaultValue={editStudent.mobile} required /></div>
   513→              <div>
   514→                <Label>Course</Label>
   515→                <Select name="courseId" defaultValue={editStudent.courseId || undefined}>
   516→                  <SelectTrigger className="w-full"><SelectValue placeholder="Select course" /></SelectTrigger>
   517→                  <SelectContent>
   518→                    {courses.map((c) => <SelectItem key={c.id} value={c.id}>{c.title}</SelectItem>)}
   519→                  </SelectContent>
   520→                </Select>
   521→              </div>
   522→              <div>
   523→                <Label>Status</Label>
   524→                <Select name="status" defaultValue={editStudent.status}>
   525→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   526→                  <SelectContent>
   527→                    <SelectItem value="PENDING">PENDING</SelectItem>
   528→                    <SelectItem value="APPROVED">APPROVED</SelectItem>
   529→                    <SelectItem value="REJECTED">REJECTED</SelectItem>
   530→                  </SelectContent>
   531→                </Select>
   532→              </div>
   533→              <DialogFooter><Button type="submit">Save Changes</Button></DialogFooter>
   534→            </form>
   535→          )}
   536→        </DialogContent>
   537→      </Dialog>

  1576→          </AlertDialogFooter>
  1577→        </AlertDialogContent>
  1578→      </AlertDialog>
  1579→    </div>

  1583→// ─── 10. Change Password Dialog ──────────────────────────────────
  1584→function ChangePasswordDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
  1585→  const [currentPassword, setCurrentPassword] = useState('')
  1586→  const [newPassword, setNewPassword] = useState('')
  1587→  const [confirmPassword, setConfirmPassword] = useState('')
  1588→  const [submitting, setSubmitting] = useState(false)
  1589→
  1590→  const handleSubmit = async (e: React.FormEvent) => {
  1591→    e.preventDefault()
  1592→    if (newPassword !== confirmPassword) {
  1593→      toast.error('New passwords do not match')
  1594→      return
  1595→    }
  1596→    if (newPassword.length < 6) {
  1597→      toast.error('Password must be at least 6 characters')
  1598→      return
  1599→    }
  1600→    setSubmitting(true)
  1601→    try {
  1602→      await api.adminChangeOwnPassword(currentPassword, newPassword)
  1603→      toast.success('Password changed successfully')
  1604→      onOpenChange(false)
  1605→      setCurrentPassword('')
  1606→      setNewPassword('')
  1607→      setConfirmPassword('')
  1608→    } catch (err: any) {
  1609→      toast.error(err.message || 'Failed to change password')
  1610→    } finally {
  1611→      setSubmitting(false)
  1612→    }
  1613→  }
  1614→
  1615→  return (
  1616→    <Dialog open={open} onOpenChange={onOpenChange}>
  1617→      <DialogContent>
  1618→        <DialogHeader>
  1619→          <DialogTitle>Change Password</DialogTitle>
  1620→          <DialogDescription>Update your admin account password</DialogDescription>
  1621→        </DialogHeader>
  1622→        <form onSubmit={handleSubmit} className="space-y-3">
  1623→          <div>
  1624→            <Label>Current Password</Label>
  1625→            <Input type="password" value={currentPassword} onChange={(e) => setCurrentPassword(e.target.value)} required />
  1626→          </div>
  1627→          <div>
  1628→            <Label>New Password</Label>
  1629→            <Input type="password" value={newPassword} onChange={(e) => setNewPassword(e.target.value)} required minLength={6} />
  1630→          </div>
  1631→          <div>
  1632→            <Label>Confirm New Password</Label>
  1633→            <Input type="password" value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} required minLength={6} />
  1634→          </div>
  1635→          <DialogFooter>
  1636→            <Button type="submit" disabled={submitting}>
  1637→              {submitting && <Loader2 className="size-4 animate-spin" />}
  1638→              Change Password
  1639→            </Button>
  1640→          </DialogFooter>
  1641→        </form>
  1642→      </DialogContent>
  1643→    </Dialog>
  1644→  )
  1645→}
