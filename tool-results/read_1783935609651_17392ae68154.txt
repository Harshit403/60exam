    17→} from 'lucide-react'

    28→// PAGE 1: DASHBOARD (with timer logic)
    29→// ═══════════════════════════════════════════════════════════════════════
    30→
    31→export function DashboardPage({ data, onRefresh }: { data: DashboardData | null; onRefresh: () => void }) {
    32→  const [selectedSubjectId, setSelectedSubjectId] = useState('')
    33→  const [selectedChapterId, setSelectedChapterId] = useState('')
    34→  const [timerPreset, setTimerPreset] = useState(0)
    35→  const [customMinutes, setCustomMinutes] = useState('')
    36→  const [timerRunning, setTimerRunning] = useState(false)
    37→  const [timerPaused, setTimerPaused] = useState(false)
    38→  const [timerSeconds, setTimerSeconds] = useState(0)
    39→  const [timerTotalSeconds, setTimerTotalSeconds] = useState(0)
    40→  const [timerCompleted, setTimerCompleted] = useState(false)
    41→  const [strikeAnimating, setStrikeAnimating] = useState(false)
    42→  const [strikeLoading, setStrikeLoading] = useState(false)
    43→  const [strikeFlash, setStrikeFlash] = useState(false)
    44→  const [resetDialogOpen, setResetDialogOpen] = useState(false)
    45→  const [resetLoading, setResetLoading] = useState(false)
    46→  const timerInterval = useRef<ReturnType<typeof setInterval> | null>(null)
    47→  const [achievementsData, setAchievementsData] = useState<any>(null)
    48→  const [showConfetti, setShowConfetti] = useState(false)
    49→  const [sessionQuote, setSessionQuote] = useState('')
    50→  const [dailyGoalMin] = useState(120)
    51→  const [sessionNotes, setSessionNotes] = useState('')
    52→  const [leaderboardData, setLeaderboardData] = useState<any[]>([])
    53→  const [leaderboardLoading, setLeaderboardLoading] = useState(true)
    54→
    55→  const dailyQuote = useMemo(() => MOTIVATIONAL_QUOTES[new Date().getDate() % MOTIVATIONAL_QUOTES.length], [])
    56→  const student = data?.student
    57→  const subjects = data?.subjects || []
    58→  const selectedSubject = subjects.find(s => s.id === selectedSubjectId)
    59→  const chapters = selectedSubject?.chapters || []
    60→
    61→  useEffect(() => {
    62→    const fetchLeaderboard = async () => {
    63→      try {
    64→        setLeaderboardLoading(true)
    65→        const data = await api.publicLeaderboard()
    66→        setLeaderboardData(data.leaderboard || [])
    67→      } catch (err) { console.error('Leaderboard fetch error:', err) }
    68→      finally { setLeaderboardLoading(false) }
    69→    }
    70→    fetchLeaderboard()
    71→  }, [])
    72→
    73→  useEffect(() => {
    74→    const fetchAchievements = async () => {
    75→      try { setAchievementsData(await api.studentAchievements()) }
    76→      catch (err) { console.error('Achievements fetch error:', err) }
    77→    }
    78→    fetchAchievements()
    79→  }, [data?.student?.score])
    80→
    81→  // Timer logic
    82→  useEffect(() => {
    83→    if (timerRunning && !timerPaused && timerSeconds > 0) {
    84→      timerInterval.current = setInterval(() => {
    85→        setTimerSeconds(prev => {
    86→          if (prev <= 1) {
    87→            setTimerRunning(false); setTimerCompleted(true); setShowConfetti(true)
    88→            if (timerInterval.current) clearInterval(timerInterval.current)
    89→            return 0
    90→          }
    91→          return prev - 1
    92→        })
    93→      }, 1000)
    94→    }
    95→    return () => { if (timerInterval.current) clearInterval(timerInterval.current) }
    96→  }, [timerRunning, timerPaused, timerSeconds])
    97→
    98→  const startTimer = () => {
    99→    let minutes = timerPreset
   100→    if (timerPreset === -1) minutes = parseInt(customMinutes) || 0
   101→    if (minutes <= 0 || !selectedChapterId) return
   102→    setTimerTotalSeconds(minutes * 60); setTimerSeconds(minutes * 60)
   103→    setTimerRunning(true); setTimerPaused(false); setTimerCompleted(false)
   104→    setSessionQuote(SESSION_QUOTES[Math.floor(Math.random() * SESSION_QUOTES.length)])
   105→  }
   106→
   107→  const resetTimer = () => {
   108→    setTimerRunning(false); setTimerPaused(false); setTimerSeconds(0)
   109→    setTimerTotalSeconds(0); setTimerCompleted(false); setSessionQuote('')
   110→    if (timerInterval.current) clearInterval(timerInterval.current)
   111→  }
   112→
   113→  const handleTimerComplete = async (completed: boolean) => {
   114→    const durationMin = Math.round(timerTotalSeconds / 60)
   115→    try {
   116→      await api.studentStartSession({ chapterId: selectedChapterId, durationMin, completed, notes: sessionNotes.trim() || undefined })
   117→      if (completed) await api.studentMarkChapter(selectedChapterId, true)
   118→    } catch (err) { console.error('Session save error:', err) }
   119→    setSessionNotes('')
   120→    setTimerCompleted(false); setShowConfetti(false); resetTimer(); onRefresh()
   121→  }
   122→
   123→  const handleStrike = async () => {
   124→    setStrikeLoading(true); setStrikeAnimating(true); setStrikeFlash(true)
   125→    try {
   126→      await api.studentSendStrike()
   127→      setTimeout(() => { setStrikeAnimating(false); setStrikeFlash(false) }, 800)
   128→      onRefresh()
   129→    } catch {
   130→      setStrikeAnimating(false); setStrikeFlash(false)
   131→    } finally { setStrikeLoading(false) }
   132→  }
   133→
   134→  const handleResetStats = async () => {
   135→    setResetLoading(true)
   136→    try { await api.studentResetStats(); setResetDialogOpen(false); onRefresh() }
   137→    catch (err) { console.error('Reset error:', err) }
   138→    finally { setResetLoading(false) }
   139→  }
   140→
   141→  if (!data || !student) return <LoadingSkeleton />
   142→
   143→  const canStrike = !(student as any).lastStrikeAt ||
   144→    new Date((student as any).lastStrikeAt).toDateString() !== new Date().toDateString()
   145→
   146→  const timerPresets = [
   147→    { label: '30m', value: 30 }, { label: '45m', value: 45 },
   148→    { label: '1h', value: 60 }, { label: '1.5h', value: 90 }, { label: '2h', value: 120 },
   149→  ]
   150→
   151→  const unlockedCount = achievementsData?.totalUnlocked || 0
   152→  const totalAchievements = achievementsData?.totalAchievements || 12
   153→  const dailyGoalProgress = Math.min((data.todayStudyMin / dailyGoalMin) * 100, 100)
   154→  const dailyGoalReached = data.todayStudyMin >= dailyGoalMin
   155→  const scoreProgress = Math.min((student.score / 1000) * 100, 100)
   156→
   157→  return (
   158→    <div className="space-y-6">
   159→      <Confetti active={showConfetti} />
   160→      {strikeFlash && <div className="fixed inset-0 bg-orange-400/20 dark:bg-orange-500/10 strike-flash pointer-events-none z-[999]" />}
   161→
   162→      {/* Daily Quote */}
   163→      <div className="rounded-xl bg-gradient-to-r from-slate-800 via-slate-700 to-slate-900 dark:from-slate-700 dark:via-slate-600 dark:to-slate-800 p-4 shadow-lg fade-in">
   164→        <div className="flex items-center gap-3">
   165→          <Sparkles className="w-5 h-5 text-amber-400 flex-shrink-0" />
   166→          <div className="min-w-0 flex-1">
   167→            <p className="text-white text-sm font-medium italic">&ldquo;{dailyQuote.text}&rdquo;</p>
   168→            <p className="text-slate-400 text-xs mt-0.5">— {dailyQuote.author}</p>
   169→          </div>
   170→        </div>
   171→      </div>
   172→
   173→      {/* Profile Card */}
   174→      <Card className="overflow-hidden">
   175→        <div className="bg-gradient-to-br from-slate-900 via-slate-800 to-slate-950 dark:from-slate-800 dark:via-slate-700 dark:to-slate-900 p-6 relative">
   176→          <div className="absolute top-0 right-0 w-40 h-40 bg-white/5 rounded-full -translate-y-1/2 translate-x-1/4" />
   177→          <div className="absolute bottom-0 left-0 w-24 h-24 bg-white/3 rounded-full translate-y-1/3 -translate-x-1/4" />
   178→          <div className="relative flex items-start justify-between">
   179→            <div className="flex items-center gap-5">
   180→              <CircularProgressRing size={80} strokeWidth={5} progress={scoreProgress} color="#f59e0b" trackColor="rgba(255,255,255,0.15)">
   181→                <Avatar className="h-14 w-14 border-2 border-white/20">
   182→                  <AvatarFallback className="bg-white/20 text-white text-xl font-bold">
   183→                    {student.fullName.charAt(0).toUpperCase()}
   184→                  </AvatarFallback>
   185→                </Avatar>
   186→              </CircularProgressRing>
   187→              <div>
   188→                <div className="flex items-center gap-2">
   189→                  <h2 className="text-xl font-bold text-white">{student.fullName}</h2>
   190→                  {student.verified && (
   191→                    <TooltipProvider><Tooltip>
   192→                      <TooltipTrigger><div className="verified-glow rounded-full p-0.5"><CheckCircle2 className="w-5 h-5 text-blue-400" /></div></TooltipTrigger>
   193→                      <TooltipContent><p>Verified - 7 Day Streak! 🔥</p></TooltipContent>
   194→                    </Tooltip></TooltipProvider>
   195→                  )}
   196→                </div>
   197→                <p className="text-slate-300 text-sm">{student.course.title}</p>
   198→                <div className="flex items-center gap-4 mt-2">
   199→                  <div className="flex items-center gap-1.5">
   200→                    <Trophy className="w-4 h-4 text-amber-400" />
   201→                    <span className="text-white font-semibold">{student.score}</span>
   202→                    <span className="text-slate-400 text-xs">points</span>
   203→                  </div>
   204→                  <div className="flex items-center gap-1.5">
   205→                    <Flame className="w-4 h-4 text-orange-400" />
   206→                    <span className="text-white font-semibold">{student.currentStreak}</span>
   207→                    <span className="text-slate-400 text-xs">day streak</span>
   208→                  </div>
   209→                </div>
   210→              </div>
   211→            </div>
   212→            <div className="flex items-center gap-2">
   213→              <TooltipProvider><Tooltip>
   214→                <TooltipTrigger asChild>
   215→                  <Button size="icon" variant="ghost"
   216→                    className={`h-11 w-11 text-orange-400 hover:text-orange-300 hover:bg-white/10 relative ${strikeAnimating ? 'fire-animate' : ''}`}
   217→                    onClick={handleStrike} disabled={!canStrike || strikeLoading}>
   218→                    <Flame className="w-6 h-6" />
   219→                    {strikeAnimating && (
   220→                      <>
   221→                        <div className="flame-particle bg-orange-500" style={{ top: '-8px', left: '6px' }} />
   222→                        <div className="flame-particle bg-red-500" style={{ top: '-12px', left: '14px', animationDelay: '0.1s' }} />
   223→                        <div className="flame-particle bg-amber-500" style={{ top: '-6px', left: '22px', animationDelay: '0.2s' }} />
   224→                        <div className="flame-particle bg-yellow-400" style={{ top: '-14px', left: '10px', animationDelay: '0.15s' }} />
   225→                        <div className="flame-particle bg-orange-400" style={{ top: '-10px', left: '18px', animationDelay: '0.25s' }} />
   226→                      </>
   227→                    )}
   228→                  </Button>
   229→                </TooltipTrigger>
   230→                <TooltipContent>{canStrike ? 'Send Strike! (+10 pts)' : 'Already struck today'}</TooltipContent>
   231→              </Tooltip></TooltipProvider>
   232→
   233→              <Dialog open={resetDialogOpen} onOpenChange={setResetDialogOpen}>
   234→                <DialogTrigger asChild>
   235→                  <Button size="icon" variant="ghost" className="h-11 w-11 text-slate-400 hover:text-white hover:bg-white/10">
   236→                    <Settings className="w-5 h-5" />
   237→                  </Button>
   238→                </DialogTrigger>
   239→                <DialogContent>
   240→                  <DialogHeader>
   241→                    <DialogTitle>Reset All Stats?</DialogTitle>
   242→                    <DialogDescription>This will permanently delete all your study sessions, plans, chapter completions, and achievements. Your score and streak will be reset to zero. This action cannot be undone.</DialogDescription>
   243→                  </DialogHeader>
   244→                  <DialogFooter>
   245→                    <Button variant="outline" onClick={() => setResetDialogOpen(false)}>Cancel</Button>
   246→                    <Button variant="destructive" onClick={handleResetStats} disabled={resetLoading}>
   247→                      {resetLoading && <Loader2 className="w-4 h-4 mr-2 animate-spin" />} Reset Stats
   248→                    </Button>
   249→                  </DialogFooter>
   250→                </DialogContent>
   251→              </Dialog>
   252→            </div>
   253→          </div>
   254→        </div>
   255→      </Card>
   256→
   257→      {/* Quick Actions */}
   258→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   259→        <Card className="cursor-pointer hover:shadow-md transition-all card-hover border-l-4 border-l-amber-500" onClick={() => {
   260→          for (const s of subjects) {
   261→            if (s.chapters.length > 0) { setSelectedSubjectId(s.id); setSelectedChapterId(s.chapters[0].id); break }
   262→          }
   263→        }}>
   264→          <CardContent className="pt-4 pb-4">
   265→            <div className="flex items-center gap-3">
   266→              <div className="p-2.5 rounded-lg bg-amber-50 dark:bg-amber-900/20">
   267→                <BookOpen className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   268→              </div>
   269→              <div>
   270→                <p className="text-sm font-semibold text-slate-900 dark:text-slate-100">Resume Last Chapter</p>
   271→                <p className="text-xs text-slate-500">Continue where you left off</p>
   272→              </div>
   273→              <ChevronRight className="w-4 h-4 text-slate-400 ml-auto" />
   274→            </div>
   275→          </CardContent>
   276→        </Card>
   277→
   278→        <Card className="cursor-pointer hover:shadow-md transition-all card-hover border-l-4 border-l-orange-500" onClick={handleStrike}>
   279→          <CardContent className="pt-4 pb-4">
   280→            <div className="flex items-center gap-3">
   281→              <div className="p-2.5 rounded-lg bg-orange-50 dark:bg-orange-900/20">
   282→                <Flame className="w-5 h-5 text-orange-600 dark:text-orange-400" />
   283→              </div>
   284→              <div>
   285→                <p className="text-sm font-semibold text-slate-900 dark:text-slate-100">Continue Streak</p>
   286→                <p className="text-xs text-slate-500">{canStrike ? 'Send your daily strike!' : 'Already struck today ✓'}</p>
   287→              </div>
   288→              <ChevronRight className="w-4 h-4 text-slate-400 ml-auto" />
   289→            </div>
   290→          </CardContent>
   291→        </Card>
   292→
   293→        <Card className="cursor-pointer hover:shadow-md transition-all card-hover border-l-4 border-l-emerald-500">
   294→          <CardContent className="pt-4 pb-4">
   295→            <div className="flex items-center gap-3">
   296→              <div className="p-2.5 rounded-lg bg-emerald-50 dark:bg-emerald-900/20">
   297→                <Target className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   298→              </div>
   299→              <div>
   300→                <p className="text-sm font-semibold text-slate-900 dark:text-slate-100">View Study Plan</p>
   301→                <p className="text-xs text-slate-500">Check today's schedule</p>
   302→              </div>
   303→              <ChevronRight className="w-4 h-4 text-slate-400 ml-auto" />
   304→            </div>
   305→          </CardContent>
   306→        </Card>
   307→      </div>
   308→
   309→      {/* Study Session + Daily Goal */}
   310→      <div className="grid md:grid-cols-[1fr_280px] gap-6">
   311→        <Card>
   312→          <CardHeader className="pb-3">
   313→            <CardTitle className="text-lg flex items-center gap-2">
   314→              <BookOpen className="w-5 h-5 text-slate-600 dark:text-slate-400" /> Study Session
   315→            </CardTitle>
   316→            <CardDescription>Which chapter do you want to study today?</CardDescription>
   317→          </CardHeader>
   318→          <CardContent className="space-y-4">
   319→            {sessionQuote && timerRunning && (
   320→              <div className="bg-gradient-to-r from-emerald-50 to-teal-50 dark:from-emerald-900/20 dark:to-teal-900/20 rounded-lg p-3 border border-emerald-200 dark:border-emerald-800 grow-in">
   321→                <p className="text-sm text-emerald-800 dark:text-emerald-200 font-medium text-center">{sessionQuote}</p>
   322→              </div>
   323→            )}
   324→
   325→            {!timerRunning && !timerCompleted ? (
   326→              <>
   327→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   328→                  <div className="space-y-1.5">
   329→                    <Label className="text-xs font-medium text-slate-500">Subject</Label>
   330→                    <Select value={selectedSubjectId} onValueChange={(v) => { setSelectedSubjectId(v); setSelectedChapterId('') }}>
   331→                      <SelectTrigger><SelectValue placeholder="Choose subject" /></SelectTrigger>
   332→                      <SelectContent>{subjects.map(s => <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>)}</SelectContent>
   333→                    </Select>
   334→                  </div>
   335→                  <div className="space-y-1.5">
   336→                    <Label className="text-xs font-medium text-slate-500">Chapter</Label>
   337→                    <Select value={selectedChapterId} onValueChange={setSelectedChapterId} disabled={!selectedSubjectId}>
   338→                      <SelectTrigger><SelectValue placeholder="Choose chapter" /></SelectTrigger>
   339→                      <SelectContent>{chapters.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}</SelectContent>
   340→                    </Select>
   341→                  </div>
   342→                </div>
   343→
   344→                <div className="space-y-2">
   345→                  <Label className="text-xs font-medium text-slate-500">Study Duration</Label>
   346→                  <div className="flex flex-wrap gap-2">
   347→                    {timerPresets.map(p => (
   348→                      <Button key={p.value} variant={timerPreset === p.value ? 'default' : 'outline'} size="sm"
   349→                        className={timerPreset === p.value ? 'bg-slate-900 dark:bg-slate-100 dark:text-slate-900' : ''}
   350→                        onClick={() => { setTimerPreset(p.value); setCustomMinutes('') }}>
   351→                        {p.label}
   352→                      </Button>
   353→                    ))}
   354→                    <div className="flex items-center gap-1.5">
   355→                      <Input type="number" placeholder="Custom (min)" className="w-28 h-8 text-sm"
   356→                        value={customMinutes} onChange={(e) => { setCustomMinutes(e.target.value); setTimerPreset(-1) }} min={1} />
   357→                      <span className="text-xs text-slate-400">min</span>
   358→                    </div>
   359→                  </div>
   360→                </div>
   361→
   362→                <Button onClick={startTimer}
   363→                  disabled={!selectedChapterId || (timerPreset === 0 && !customMinutes) || (timerPreset === -1 && !customMinutes)}
   364→                  className="w-full sm:w-auto bg-slate-900 hover:bg-slate-800 dark:bg-slate-100 dark:text-slate-900 dark:hover:bg-slate-200">
   365→                  <Play className="w-4 h-4 mr-2" /> Start Session
   366→                </Button>
   367→              </>
   368→            ) : timerCompleted ? (
   369→              <div className="text-center py-6 space-y-4">
   370→                <div className="mx-auto w-16 h-16 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center celebration-burst">
   371→                  <CheckCircle2 className="w-8 h-8 text-emerald-600 dark:text-emerald-400" />
   372→                </div>
   373→                <h3 className="text-lg font-semibold text-slate-900 dark:text-slate-100">Session Complete! 🎉</h3>
   374→                <p className="text-slate-500 dark:text-slate-400">Have you completed this chapter?</p>
   375→                {/* Session Notes Input */}
   376→                <div className="text-left space-y-1.5 max-w-md mx-auto">
   377→                  <label className="text-xs font-medium text-slate-600 dark:text-slate-400 flex items-center gap-1.5">
   378→                    <StickyNote className="w-3 h-3" /> Session Notes (optional)
   379→                  </label>
   380→                  <textarea
   381→                    placeholder="What did you learn? Any key takeaways?"
   382→                    value={sessionNotes}
   383→                    onChange={(e) => setSessionNotes(e.target.value)}
   384→                    rows={3}
   385→                    className="w-full px-3 py-2 text-sm rounded-lg border border-slate-200 dark:border-slate-800 bg-white dark:bg-slate-900 focus:outline-none focus:ring-2 focus:ring-slate-300 dark:focus:ring-slate-700 resize-none"
   386→                  />
   387→                </div>
   388→                <div className="flex items-center justify-center gap-3">
   389→                  <Button onClick={() => handleTimerComplete(true)} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   390→                    <CheckCircle2 className="w-4 h-4 mr-2" /> Yes, Done!
   391→                  </Button>
   392→                  <Button variant="outline" onClick={() => handleTimerComplete(false)}>Not Yet</Button>
   393→                </div>
   394→              </div>
   395→            ) : (
   396→              <div className="text-center py-6 space-y-4">
   397→                <div>
   398→                  <p className="text-sm text-slate-500 dark:text-slate-400 mb-1">Studying</p>
   399→                  <p className="font-medium text-slate-900 dark:text-slate-100">{chapters.find(c => c.id === selectedChapterId)?.name}</p>
   400→                </div>
   401→                <div className="relative inline-flex items-center justify-center">
   402→                  <TimerCountdownRing size={200} strokeWidth={8} totalSeconds={timerTotalSeconds} remainingSeconds={timerSeconds} />
   403→                  <div className="absolute inset-0 flex flex-col items-center justify-center">
   404→                    <span className="text-4xl font-mono font-bold text-slate-900 dark:text-slate-100 tabular-nums">{formatTimer(timerSeconds)}</span>
   405→                    <span className="text-xs text-slate-400 mt-1">{timerPaused ? 'PAUSED' : 'remaining'}</span>
   406→                  </div>
   407→                </div>
   408→                <div className="flex items-center justify-center gap-3">
   409→                  <Button variant="outline" size="icon" onClick={() => setTimerPaused(prev => !prev)} className="h-12 w-12 rounded-full">
