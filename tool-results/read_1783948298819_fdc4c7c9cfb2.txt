   185→function InlineLineChart({ data, height=200 }: { data:{day:string;signups:number}[]; height?:number }) {
   186→  if (!data.length) return null
   187→  const maxVal = Math.max(...data.map(d=>d.signups), 1)
   188→  const w = 500, h = height, padL = 40, padR = 20, padT = 20, padB = 40
   189→  const plotW = w - padL - padR, plotH = h - padT - padB
   190→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   191→  const xScale = (i:number) => padL + (i/(data.length-1))*plotW
   192→
   193→  const linePath = data.map((d,i)=>`${i===0?'M':'L'}${xScale(i)},${yScale(d.signups)}`).join(' ')
   194→  const areaPath = linePath + ` L${xScale(data.length-1)},${padT+plotH} L${padL},${padT+plotH} Z`
   195→  const yTicks = 5
   196→  const yStep = maxVal / yTicks
   197→
   198→  return (
   199→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   200→      {/* Grid lines */}
   201→      {Array.from({length:yTicks+1}).map((_,i)=>{
   202→        const val = Math.round(i*yStep)
   203→        const y = yScale(val)
   204→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   205→      })}
   206→      {/* Area fill */}
   207→      <path d={areaPath} fill="#10b981" fillOpacity={0.08} />
   208→      {/* Line */}
   209→      <path d={linePath} fill="none" stroke="#10b981" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
   210→      {/* Dots + labels */}
   211→      {data.map((d,i)=><g key={i}>
   212→        <circle cx={xScale(i)} cy={yScale(d.signups)} r="4" fill="#10b981" />
   213→        <text x={xScale(i)} y={h-8} textAnchor="middle" className="fill-muted-foreground" fontSize="11">{d.day}</text>
   214→      </g>)}
   215→    </svg>
   216→  )
   217→}

   219→function InlineBarChart({ data, height=220 }: { data:TopPerformer[]; height?:number }) {
   220→  if (!data.length) return null
   221→  const maxVal = Math.max(...data.map(d=>d.studyHours), 1)
   222→  const barW = Math.min(40, 400/data.length - 8)
   223→  const w = 500, h = height, padL = 40, padR = 20, padT = 10, padB = 60
   224→  const plotW = w - padL - padR, plotH = h - padT - padB
   225→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   226→  const gap = plotW / data.length
   227→
   228→  return (
   229→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>

   251→// ─── Hooks ───────────────────────────────────────────────────────
   252→function useFetch<T>(fetcher:()=>Promise<T>, deps:any[]) {
   253→  const [data, setData] = useState<T|null>(null)
   254→  const [loading, setLoading] = useState(true)
   255→  const [error, setError] = useState<string|null>(null)
   256→  const [refreshKey, setRefreshKey] = useState(0)
   257→  const refresh = useCallback(()=>setRefreshKey(k=>k+1), [])
   258→  useEffect(()=>{
   259→    let c=false
   260→    fetcher().then(r=>{if(c)return;setData(r);setError(null);setLoading(false)}).catch(e=>{if(c)return;setError(e.message||'Failed');setLoading(false)})
   261→    return()=>{c=true}
   262→  }, [...deps, refreshKey])
   263→  return { data, loading, error, refresh, setData }
   264→}

   266→function usePagination<T>(items:T[], pageSize=10) {
   267→  const [page, setPage] = useState(1)
   268→  const safeItems = Array.isArray(items) ? items : []
   269→  const totalPages = Math.max(1, Math.ceil(safeItems.length/pageSize))
   270→  const safePage = Math.min(page, totalPages)
   271→  const paginated = safeItems.slice((safePage-1)*pageSize, safePage*pageSize)
   272→  const goTo = (p:number)=>setPage(Math.max(1,Math.min(p,totalPages)))
   273→  if (page>totalPages&&totalPages>0) setPage(totalPages)
   274→  return { page:safePage, totalPages, paginated, goTo, setPage }
   275→}
