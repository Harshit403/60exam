    91→// Animated counter hook
    92→function useAnimatedCounter(end: number, duration: number = 2000, startOnView: boolean = true) {
    93→  const [count, setCount] = useState(0)
    94→  const ref = useRef<HTMLSpanElement>(null)
    95→  const isInView = useInView(ref, { once: true })
    96→  const hasStarted = useRef(false)
    97→
    98→  useEffect(() => {
    99→    if (startOnView && !isInView) return
   100→    if (hasStarted.current) return
   101→    hasStarted.current = true
   102→
   103→    let startTime: number | null = null
   104→    const step = (timestamp: number) => {
   105→      if (!startTime) startTime = timestamp
   106→      const progress = Math.min((timestamp - startTime) / duration, 1)
   107→      setCount(Math.floor(progress * end))
   108→      if (progress < 1) {
   109→        requestAnimationFrame(step)
   110→      } else {
   111→        setCount(end)
   112→      }
   113→    }
   114→    requestAnimationFrame(step)
   115→  }, [end, duration, isInView, startOnView])
   116→
   117→  return { count, ref }
   118→}
