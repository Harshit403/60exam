    45→const timeAgo = (dateStr: string) => {
    46→  const now = new Date()
    47→  const date = new Date(dateStr)
    48→  const diffMs = now.getTime() - date.getTime()
    49→  const diffMin = Math.floor(diffMs / 60000)
    50→  const diffHrs = Math.floor(diffMin / 60)
    51→  const diffDays = Math.floor(diffHrs / 24)
    52→  if (diffMin < 1) return 'just now'
    53→  if (diffMin < 60) return `${diffMin}m ago`
    54→  if (diffHrs < 24) return `${diffHrs}h ago`
    55→  if (diffDays < 7) return `${diffDays}d ago`
    56→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
    57→}
