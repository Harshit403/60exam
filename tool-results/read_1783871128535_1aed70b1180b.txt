   175→   175→// ─── Custom hook: data fetcher with refresh ──────────────────────
   176→   176→function useFetch<T>(fetcher: () => Promise<T>, deps: any[]) {
   177→   177→  const [data, setData] = useState<T | null>(null)
   178→   178→  const [loading, setLoading] = useState(true)
   179→   179→  const [error, setError] = useState<string | null>(null)
   180→   180→  const [refreshKey, setRefreshKey] = useState(0)
   181→   181→
   182→   182→  const refresh = useCallback(() => setRefreshKey((k) => k + 1), [])
   183→   183→
   184→   184→  useEffect(() => {
   185→   185→    let cancelled = false
   186→   186→    fetcher()
   187→   187→      .then((res) => {
   188→   188→        if (cancelled) return
   189→   189→        setData(res)
   190→   190→        setError(null)
   191→   191→        setLoading(false)
   192→   192→      })
   193→   193→      .catch((err) => {
   194→   194→        if (cancelled) return
   195→   195→        setError(err.message || 'Failed to fetch')
   196→   196→        setLoading(false)
   197→   197→      })
   198→   198→    return () => { cancelled = true }
   199→   199→  }, [...deps, refreshKey])
   200→   200→
   201→   201→  return { data, loading, error, refresh, setData }
   202→   202→}
