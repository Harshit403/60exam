    29→
    30→export function DashboardPage({ data, onRefresh }: { data: DashboardData | null; onRefresh: () => void }) {
    31→  const [selectedSubjectId, setSelectedSubjectId] = useState('')
    32→  const [selectedChapterId, setSelectedChapterId] = useState('')
    33→  const [timerPreset, setTimerPreset] = useState(0)
    34→  const [customMinutes, setCustomMinutes] = useState('')
    35→  const [timerRunning, setTimerRunning] = useState(false)
    36→  const [timerPaused, setTimerPaused] = useState(false)
    37→  const [timerSeconds, setTimerSeconds] = useState(0)
    38→  const [timerTotalSeconds, setTimerTotalSeconds] = useState(0)
    39→  const [timerCompleted, setTimerCompleted] = useState(false)
    40→  const [strikeAnimating, setStrikeAnimating] = useState(false)
    41→  const [strikeLoading, setStrikeLoading] = useState(false)
    42→  const [strikeFlash, setStrikeFlash] = useState(false)
    43→  const [resetDialogOpen, setResetDialogOpen] = useState(false)
    44→  const [resetLoading, setResetLoading] = useState(false)
    45→  const timerInterval = useRef<ReturnType<typeof setInterval> | null>(null)
    46→  const [achievementsData, setAchievementsData] = useState<any>(null)
    47→  const [showConfetti, setShowConfetti] = useState(false)
    48→  const [sessionQuote, setSessionQuote] = useState('')
    49→  const [dailyGoalMin] = useState(120)
    50→  const [sessionNotes, setSessionNotes] = useState('')
    51→
    52→  const dailyQuote = useMemo(() => MOTIVATIONAL_QUOTES[new Date().getDate() % MOTIVATIONAL_QUOTES.length], [])
    53→  const student = data?.student
    54→  const subjects = data?.subjects || []
    55→  const selectedSubject = subjects.find(s => s.id === selectedSubjectId)
    56→  const chapters = selectedSubject?.chapters || []
    57→
    58→  useEffect(() => {
    59→    const fetchAchievements = async () => {
    60→      try { setAchievementsData(await api.studentAchievements()) }
    61→      catch (err) { console.error('Achievements fetch error:', err) }
    62→    }
    63→    fetchAchievements()
    64→  }, [data?.student?.score])
    65→
    66→  // Timer logic
    67→  useEffect(() => {
    68→    if (timerRunning && !timerPaused && timerSeconds > 0) {
    69→      timerInterval.current = setInterval(() => {
    70→        setTimerSeconds(prev => {
    71→          if (prev <= 1) {
    72→            setTimerRunning(false); setTimerCompleted(true); setShowConfetti(true)
    73→            if (timerInterval.current) clearInterval(timerInterval.current)
    74→            return 0
    75→          }
    76→          return prev - 1
    77→        })
    78→      }, 1000)
    79→    }
    80→    return () => { if (timerInterval.current) clearInterval(timerInterval.current) }
    81→  }, [timerRunning, timerPaused, timerSeconds])
    82→
    83→  const startTimer = () => {
    84→    let minutes = timerPreset
    85→    if (timerPreset === -1) minutes = parseInt(customMinutes) || 0
    86→    if (minutes <= 0 || !selectedChapterId) return
    87→    setTimerTotalSeconds(minutes * 60); setTimerSeconds(minutes * 60)
    88→    setTimerRunning(true); setTimerPaused(false); setTimerCompleted(false)
    89→    setSessionQuote(SESSION_QUOTES[Math.floor(Math.random() * SESSION_QUOTES.length)])
    90→  }
    91→
    92→  const resetTimer = () => {
    93→    setTimerRunning(false); setTimerPaused(false); setTimerSeconds(0)
    94→    setTimerTotalSeconds(0); setTimerCompleted(false); setSessionQuote('')
    95→    if (timerInterval.current) clearInterval(timerInterval.current)
    96→  }
    97→
    98→  const handleTimerComplete = async (completed: boolean) => {
    99→    const durationMin = Math.round(timerTotalSeconds / 60)
   100→    try {
   101→      await api.studentStartSession({ chapterId: selectedChapterId, durationMin, completed, notes: sessionNotes.trim() || undefined })
   102→      if (completed) await api.studentMarkChapter(selectedChapterId, true)
   103→    } catch (err) { console.error('Session save error:', err) }
   104→    setSessionNotes('')
   105→    setTimerCompleted(false); setShowConfetti(false); resetTimer(); onRefresh()
   106→  }
   107→
   108→  const handleStrike = async () => {
   109→    setStrikeLoading(true); setStrikeAnimating(true); setStrikeFlash(true)
   110→    try {
   111→      await api.studentSendStrike()
   112→      setTimeout(() => { setStrikeAnimating(false); setStrikeFlash(false) }, 800)
   113→      onRefresh()
   114→    } catch {
   115→      setStrikeAnimating(false); setStrikeFlash(false)
   116→    } finally { setStrikeLoading(false) }
   117→  }
   118→
   119→  const handleResetStats = async () => {
   120→    setResetLoading(true)
   121→    try { await api.studentResetStats(); setResetDialogOpen(false); onRefresh() }
   122→    catch (err) { console.error('Reset error:', err) }
   123→    finally { setResetLoading(false) }
   124→  }
   125→
   126→  if (!data || !student) return <LoadingSkeleton />
   127→
   128→  const canStrike = !(student as any).lastStrikeAt ||
   129→    new Date((student as any).lastStrikeAt).toDateString() !== new Date().toDateString()
   130→
   131→  const timerPresets = [
   132→    { label: '30m', value: 30 }, { label: '45m', value: 45 },
   133→    { label: '1h', value: 60 }, { label: '1.5h', value: 90 }, { label: '2h', value: 120 },
   134→  ]
   135→
   136→  const unlockedCount = achievementsData?.totalUnlocked || 0
   137→  const totalAchievements = achievementsData?.totalAchievements || 12
   138→  const dailyGoalProgress = Math.min((data.todayStudyMin / dailyGoalMin) * 100, 100)
   139→  const dailyGoalReached = data.todayStudyMin >= dailyGoalMin
   140→  const scoreProgress = Math.min((student.score / 1000) * 100, 100)
   141→
   142→  return (
   143→    <div className="space-y-6">
   144→      <Confetti active={showConfetti} />
   145→      {strikeFlash && <div className="fixed inset-0 bg-orange-400/20 dark:bg-orange-500/10 strike-flash pointer-events-none z-[999]" />}
   146→
   147→      {/* Daily Quote */}
   148→      <div className="rounded-xl bg-gradient-to-r from-slate-800 via-slate-700 to-slate-900 dark:from-slate-700 dark:via-slate-600 dark:to-slate-800 p-4 shadow-lg fade-in">
   149→        <div className="flex items-center gap-3">
   150→          <Sparkles className="w-5 h-5 text-amber-400 flex-shrink-0" />
   151→          <div className="min-w-0 flex-1">
   152→            <p className="text-white text-sm font-medium italic">&ldquo;{dailyQuote.text}&rdquo;</p>
   153→            <p className="text-slate-400 text-xs mt-0.5">— {dailyQuote.author}</p>
   154→          </div>
   155→        </div>
   156→      </div>
   157→
   158→      {/* Profile Card */}
   159→      <Card className="overflow-hidden">
   160→        <div className="bg-gradient-to-br from-slate-900 via-slate-800 to-slate-950 dark:from-slate-800 dark:via-slate-700 dark:to-slate-900 p-6 relative">
   161→          <div className="absolute top-0 right-0 w-40 h-40 bg-white/5 rounded-full -translate-y-1/2 translate-x-1/4" />
   162→          <div className="absolute bottom-0 left-0 w-24 h-24 bg-white/3 rounded-full translate-y-1/3 -translate-x-1/4" />
   163→          <div className="relative flex items-start justify-between">
   164→            <div className="flex items-center gap-5">
   165→              <CircularProgressRing size={80} strokeWidth={5} progress={scoreProgress} color="#f59e0b" trackColor="rgba(255,255,255,0.15)">
   166→                <Avatar className="h-14 w-14 border-2 border-white/20">
   167→                  <AvatarFallback className="bg-white/20 text-white text-xl font-bold">
   168→                    {student.fullName.charAt(0).toUpperCase()}
   169→                  </AvatarFallback>
   170→                </Avatar>
   171→              </CircularProgressRing>
   172→              <div>
   173→                <div className="flex items-center gap-2">
   174→                  <h2 className="text-xl font-bold text-white">{student.fullName}</h2>
   175→                  {student.verified && (
   176→                    <TooltipProvider><Tooltip>
   177→                      <TooltipTrigger><div className="verified-glow rounded-full p-0.5"><CheckCircle2 className="w-5 h-5 text-blue-400" /></div></TooltipTrigger>
   178→                      <TooltipContent><p>Verified - 7 Day Streak! 🔥</p></TooltipContent>
   179→                    </Tooltip></TooltipProvider>
   180→                  )}
   181→                </div>
   182→                <p className="text-slate-300 text-sm">{student.course.title}</p>
   183→                <div className="flex items-center gap-4 mt-2">
   184→                  <div className="flex items-center gap-1.5">
   185→                    <Trophy className="w-4 h-4 text-amber-400" />
   186→                    <span className="text-white font-semibold">{student.score}</span>
   187→                    <span className="text-slate-400 text-xs">points</span>
   188→                  </div>
   189→                  <div className="flex items-center gap-1.5">
   190→                    <Flame className="w-4 h-4 text-orange-400" />
   191→                    <span className="text-white font-semibold">{student.currentStreak}</span>
   192→                    <span className="text-slate-400 text-xs">day streak</span>
   193→                  </div>
   194→                </div>
   195→              </div>
   196→            </div>
   197→            <div className="flex items-center gap-2">
   198→              <TooltipProvider><Tooltip>
   199→                <TooltipTrigger asChild>
   200→                  <Button size="icon" variant="ghost"
   201→                    className={`h-11 w-11 text-orange-400 hover:text-orange-300 hover:bg-white/10 relative ${strikeAnimating ? 'fire-animate' : ''}`}
   202→                    onClick={handleStrike} disabled={!canStrike || strikeLoading}>
   203→                    <Flame className="w-6 h-6" />
   204→                    {strikeAnimating && (
   205→                      <>
   206→                        <div className="flame-particle bg-orange-500" style={{ top: '-8px', left: '6px' }} />
   207→                        <div className="flame-particle bg-red-500" style={{ top: '-12px', left: '14px', animationDelay: '0.1s' }} />
   208→                        <div className="flame-particle bg-amber-500" style={{ top: '-6px', left: '22px', animationDelay: '0.2s' }} />
   209→                        <div className="flame-particle bg-yellow-400" style={{ top: '-14px', left: '10px', animationDelay: '0.15s' }} />
   210→                        <div className="flame-particle bg-orange-400" style={{ top: '-10px', left: '18px', animationDelay: '0.25s' }} />
   211→                      </>
   212→                    )}
   213→                  </Button>
   214→                </TooltipTrigger>
   215→                <TooltipContent>{canStrike ? 'Send Strike! (+10 pts)' : 'Already struck today'}</TooltipContent>
   216→              </Tooltip></TooltipProvider>
   217→
   218→              <Dialog open={resetDialogOpen} onOpenChange={setResetDialogOpen}>
   219→                <DialogTrigger asChild>
   220→                  <Button size="icon" variant="ghost" className="h-11 w-11 text-slate-400 hover:text-white hover:bg-white/10">
   221→                    <Settings className="w-5 h-5" />
   222→                  </Button>
   223→                </DialogTrigger>
   224→                <DialogContent>
   225→                  <DialogHeader>
   226→                    <DialogTitle>Reset All Stats?</DialogTitle>
   227→                    <DialogDescription>This will permanently delete all your study sessions, plans, chapter completions, and achievements. Your score and streak will be reset to zero. This action cannot be undone.</DialogDescription>
   228→                  </DialogHeader>
   229→                  <DialogFooter>
   230→                    <Button variant="outline" onClick={() => setResetDialogOpen(false)}>Cancel</Button>
   231→                    <Button variant="destructive" onClick={handleResetStats} disabled={resetLoading}>
   232→                      {resetLoading && <Loader2 className="w-4 h-4 mr-2 animate-spin" />} Reset Stats
   233→                    </Button>
   234→                  </DialogFooter>
   235→                </DialogContent>
   236→              </Dialog>
   237→            </div>
   238→          </div>
   239→        </div>
   240→      </Card>
   241→
   242→      {/* Quick Actions */}
   243→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   244→        <Card className="cursor-pointer hover:shadow-md transition-all card-hover border-l-4 border-l-amber-500" onClick={() => {
   245→          for (const s of subjects) {
   246→            if (s.chapters.length > 0) { setSelectedSubjectId(s.id); setSelectedChapterId(s.chapters[0].id); break }
   247→          }
   248→        }}>
   249→          <CardContent className="pt-4 pb-4">
   250→            <div className="flex items-center gap-3">
   251→              <div className="p-2.5 rounded-lg bg-amber-50 dark:bg-amber-900/20">
   252→                <BookOpen className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   253→              </div>
   254→              <div>
   255→                <p className="text-sm font-semibold text-slate-900 dark:text-slate-100">Resume Last Chapter</p>
   256→                <p className="text-xs text-slate-500">Continue where you left off</p>
   257→              </div>
   258→              <ChevronRight className="w-4 h-4 text-slate-400 ml-auto" />
   259→            </div>
   260→          </CardContent>
   261→        </Card>
   262→
   263→        <Card className="cursor-pointer hover:shadow-md transition-all card-hover border-l-4 border-l-orange-500" onClick={handleStrike}>
   264→          <CardContent className="pt-4 pb-4">
   265→            <div className="flex items-center gap-3">
   266→              <div className="p-2.5 rounded-lg bg-orange-50 dark:bg-orange-900/20">
   267→                <Flame className="w-5 h-5 text-orange-600 dark:text-orange-400" />
   268→              </div>
   269→              <div>
   270→                <p className="text-sm font-semibold text-slate-900 dark:text-slate-100">Continue Streak</p>
   271→                <p className="text-xs text-slate-500">{canStrike ? 'Send your daily strike!' : 'Already struck today ✓'}</p>
   272→              </div>
   273→              <ChevronRight className="w-4 h-4 text-slate-400 ml-auto" />
   274→            </div>
   275→          </CardContent>
   276→        </Card>
   277→
   278→        <Card className="cursor-pointer hover:shadow-md transition-all card-hover border-l-4 border-l-emerald-500">
   279→          <CardContent className="pt-4 pb-4">
   280→            <div className="flex items-center gap-3">
   281→              <div className="p-2.5 rounded-lg bg-emerald-50 dark:bg-emerald-900/20">
   282→                <Target className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   283→              </div>
   284→              <div>
   285→                <p className="text-sm font-semibold text-slate-900 dark:text-slate-100">View Study Plan</p>
   286→                <p className="text-xs text-slate-500">Check today's schedule</p>
   287→              </div>
   288→              <ChevronRight className="w-4 h-4 text-slate-400 ml-auto" />
   289→            </div>
   290→          </CardContent>
   291→        </Card>
   292→      </div>
   293→
   294→      {/* Study Session + Daily Goal */}
   295→      <div className="grid md:grid-cols-[1fr_280px] gap-6">
   296→        <Card>
   297→          <CardHeader className="pb-3">
   298→            <CardTitle className="text-lg flex items-center gap-2">
   299→              <BookOpen className="w-5 h-5 text-slate-600 dark:text-slate-400" /> Study Session
   300→            </CardTitle>
   301→            <CardDescription>Which chapter do you want to study today?</CardDescription>
   302→          </CardHeader>
   303→          <CardContent className="space-y-4">
   304→            {sessionQuote && timerRunning && (
   305→              <div className="bg-gradient-to-r from-emerald-50 to-teal-50 dark:from-emerald-900/20 dark:to-teal-900/20 rounded-lg p-3 border border-emerald-200 dark:border-emerald-800 grow-in">
   306→                <p className="text-sm text-emerald-800 dark:text-emerald-200 font-medium text-center">{sessionQuote}</p>
   307→              </div>
   308→            )}
   309→
   310→            {!timerRunning && !timerCompleted ? (
   311→              <>
   312→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   313→                  <div className="space-y-1.5">
   314→                    <Label className="text-xs font-medium text-slate-500">Subject</Label>
   315→                    <Select value={selectedSubjectId} onValueChange={(v) => { setSelectedSubjectId(v); setSelectedChapterId('') }}>
   316→                      <SelectTrigger><SelectValue placeholder="Choose subject" /></SelectTrigger>
   317→                      <SelectContent>{subjects.map(s => <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>)}</SelectContent>
   318→                    </Select>
   319→                  </div>
   320→                  <div className="space-y-1.5">
   321→                    <Label className="text-xs font-medium text-slate-500">Chapter</Label>
   322→                    <Select value={selectedChapterId} onValueChange={setSelectedChapterId} disabled={!selectedSubjectId}>
   323→                      <SelectTrigger><SelectValue placeholder="Choose chapter" /></SelectTrigger>
   324→                      <SelectContent>{chapters.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}</SelectContent>
   325→                    </Select>
   326→                  </div>
   327→                </div>
   328→
   329→                <div className="space-y-2">
   330→                  <Label className="text-xs font-medium text-slate-500">Study Duration</Label>
   331→                  <div className="flex flex-wrap gap-2">
   332→                    {timerPresets.map(p => (
   333→                      <Button key={p.value} variant={timerPreset === p.value ? 'default' : 'outline'} size="sm"
   334→                        className={timerPreset === p.value ? 'bg-slate-900 dark:bg-slate-100 dark:text-slate-900' : ''}
   335→                        onClick={() => { setTimerPreset(p.value); setCustomMinutes('') }}>
   336→                        {p.label}
   337→                      </Button>
   338→                    ))}
   339→                    <div className="flex items-center gap-1.5">
   340→                      <Input type="number" placeholder="Custom (min)" className="w-28 h-8 text-sm"
   341→                        value={customMinutes} onChange={(e) => { setCustomMinutes(e.target.value); setTimerPreset(-1) }} min={1} />
   342→                      <span className="text-xs text-slate-400">min</span>
   343→                    </div>
   344→                  </div>
   345→                </div>
   346→
   347→                <Button onClick={startTimer}
   348→                  disabled={!selectedChapterId || (timerPreset === 0 && !customMinutes) || (timerPreset === -1 && !customMinutes)}
   349→                  className="w-full sm:w-auto bg-slate-900 hover:bg-slate-800 dark:bg-slate-100 dark:text-slate-900 dark:hover:bg-slate-200">
   350→                  <Play className="w-4 h-4 mr-2" /> Start Session
   351→                </Button>
   352→              </>
   353→            ) : timerCompleted ? (
   354→              <div className="text-center py-6 space-y-4">
   355→                <div className="mx-auto w-16 h-16 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center celebration-burst">
   356→                  <CheckCircle2 className="w-8 h-8 text-emerald-600 dark:text-emerald-400" />
   357→                </div>
   358→                <h3 className="text-lg font-semibold text-slate-900 dark:text-slate-100">Session Complete! 🎉</h3>
   359→                <p className="text-slate-500 dark:text-slate-400">Have you completed this chapter?</p>
   360→                {/* Session Notes Input */}
   361→                <div className="text-left space-y-1.5 max-w-md mx-auto">
   362→                  <label className="text-xs font-medium text-slate-600 dark:text-slate-400 flex items-center gap-1.5">
   363→                    <StickyNote className="w-3 h-3" /> Session Notes (optional)
   364→                  </label>
   365→                  <textarea
   366→                    placeholder="What did you learn? Any key takeaways?"
   367→                    value={sessionNotes}
   368→                    onChange={(e) => setSessionNotes(e.target.value)}
   369→                    rows={3}
   370→                    className="w-full px-3 py-2 text-sm rounded-lg border border-slate-200 dark:border-slate-800 bg-white dark:bg-slate-900 focus:outline-none focus:ring-2 focus:ring-slate-300 dark:focus:ring-slate-700 resize-none"
   371→                  />
   372→                </div>
   373→                <div className="flex items-center justify-center gap-3">
   374→                  <Button onClick={() => handleTimerComplete(true)} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   375→                    <CheckCircle2 className="w-4 h-4 mr-2" /> Yes, Done!
   376→                  </Button>
   377→                  <Button variant="outline" onClick={() => handleTimerComplete(false)}>Not Yet</Button>
   378→                </div>
   379→              </div>
   380→            ) : (
   381→              <div className="text-center py-6 space-y-4">
   382→                <div>
   383→                  <p className="text-sm text-slate-500 dark:text-slate-400 mb-1">Studying</p>

   385→                </div>
   386→                <div className="relative inline-flex items-center justify-center">
   387→                  <TimerCountdownRing size={200} strokeWidth={8} totalSeconds={timerTotalSeconds} remainingSeconds={timerSeconds} />
   388→                  <div className="absolute inset-0 flex flex-col items-center justify-center">
   389→                    <span className="text-4xl font-mono font-bold text-slate-900 dark:text-slate-100 tabular-nums">{formatTimer(timerSeconds)}</span>

   391→                  </div>
   392→                </div>
   393→                <div className="flex items-center justify-center gap-3">
   394→                  <Button variant="outline" size="icon" onClick={() => setTimerPaused(prev => !prev)} className="h-12 w-12 rounded-full">

   396→                  </Button>
   397→                  <Button variant="outline" size="icon" onClick={resetTimer} className="h-12 w-12 rounded-full">
   398→                    <RotateCcw className="w-5 h-5" />
   399→                  </Button>
   400→                </div>
   401→                <Button variant="ghost" onClick={() => { resetTimer(); setSelectedChapterId('') }} className="text-sm text-slate-500">
   402→                  Choose Another
   403→                </Button>
   404→              </div>

   406→          </CardContent>
   407→        </Card>
   408→
   409→        {/* Right Column */}
   410→        <div className="space-y-6">
