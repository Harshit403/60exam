   490→function RecentTrendChart({ trend }: { trend: { date: string; attempts: number; avgScore: number }[] }) {
   491→  const maxScore = 100
   492→  const hasData = trend.some((d) => d.attempts > 0)
   493→  const maxAttempts = Math.max(...trend.map((d) => d.attempts), 1)
   494→
   495→  if (!hasData) {
   496→    return (
   497→      <div className="py-8 text-center">
   498→        <Calendar className="w-8 h-8 mx-auto text-slate-300 dark:text-slate-700 mb-2" />
   499→        <p className="text-xs text-slate-500 dark:text-slate-400">No quiz attempts in the last 7 days</p>
   500→      </div>
   501→    )
   502→  }
   503→
   504→  return (
   505→    <div className="space-y-2">
   506→      <div className="flex items-end gap-1.5 h-32">
   507→        {trend.map((d, i) => {
   508→          const heightPct = (d.avgScore / maxScore) * 100
   509→          const hasAttempts = d.attempts > 0
   510→          const intensity = d.attempts / maxAttempts
   511→          // Color intensity based on score: higher score = deeper emerald
   512→          const barColor = !hasAttempts
   513→            ? 'bg-slate-100 dark:bg-slate-800'
   514→            : d.avgScore >= 75
   515→              ? 'bg-gradient-to-t from-emerald-600 to-emerald-400'
   516→              : d.avgScore >= 50
   517→                ? 'bg-gradient-to-t from-amber-500 to-amber-400'
   518→                : 'bg-gradient-to-t from-rose-500 to-rose-400'
   519→          return (
   520→            <div key={i} className="flex-1 group relative flex flex-col items-center">
   521→              <div className="w-full h-full flex items-end">
   522→                <div
   523→                  className={`w-full rounded-t transition-all duration-300 ${barColor}`}
   524→                  style={{
   525→                    height: `${hasAttempts ? Math.max(heightPct, 8) : 0}%`,
   526→                    minHeight: hasAttempts ? '6px' : '0',
   527→                    opacity: hasAttempts ? 0.4 + intensity * 0.6 : 1,
   528→                  }}
   529→                />
   530→              </div>
   531→              {/* Tooltip */}
   532→              <div className="absolute bottom-full mb-1 left-1/2 -translate-x-1/2 bg-slate-900 text-white text-[10px] px-2 py-1 rounded opacity-0 group-hover:opacity-100 pointer-events-none whitespace-nowrap z-10">
   533→                {formatShortDate(d.date)}: {hasAttempts ? `${d.avgScore}% · ${d.attempts} attempt(s)` : 'No attempts'}
   534→              </div>
   535→            </div>
   536→          )
   537→        })}
   538→      </div>
   539→      <div className="flex justify-between text-[10px] text-slate-400">
   540→        {trend.map((d, i) => (
   541→          <span key={i} className="flex-1 text-center">
   542→            {new Date(d.date + 'T00:00:00').toLocaleDateString('en-US', { weekday: 'short' }).charAt(0)}

   544→        ))}
   545→      </div>
   546→      <div className="flex items-center justify-between text-[10px] text-slate-400 pt-1 border-t border-slate-100 dark:border-slate-800">
   547→        <span>Range: 0–100%</span>
   548→        <span>{trend.reduce((s, d) => s + d.attempts, 0)} total attempts</span>
   549→      </div>
   550→    </div>
   551→  )

   554→function AttemptCard({ attempt, index }: { attempt: QuizAttemptItem; index: number }) {

   556→  const isPass = attempt.passed
   557→

   562→      return Array.isArray(parsed) ? (parsed as number[]) : []

   564→      return []
   565→    }
   566→  }, [attempt.answers])

   574→        isPass
