   175→// ─── Custom hook: data fetcher with refresh ──────────────────────
   176→function useFetch<T>(fetcher: () => Promise<T>, deps: any[]) {
   177→  const [data, setData] = useState<T | null>(null)
   178→  const [loading, setLoading] = useState(true)
   179→  const [error, setError] = useState<string | null>(null)
   180→  const [refreshKey, setRefreshKey] = useState(0)
   181→
   182→  const refresh = useCallback(() => setRefreshKey((k) => k + 1), [])
   183→
   184→  useEffect(() => {
   185→    let cancelled = false
   186→    fetcher()
   187→      .then((res) => {
   188→        if (cancelled) return
   189→        setData(res)
   190→        setError(null)
   191→        setLoading(false)
   192→      })
   193→      .catch((err) => {
   194→        if (cancelled) return
   195→        setError(err.message || 'Failed to fetch')
   196→        setLoading(false)
   197→      })
   198→    return () => { cancelled = true }
   199→  }, [...deps, refreshKey])
   200→
   201→  return { data, loading, error, refresh, setData }
   202→}

   375→// ─── 2. Students Page ────────────────────────────────────────────
   376→function StudentsPage() {
   377→  const [search, setSearch] = useState('')
   378→  const [addOpen, setAddOpen] = useState(false)
   379→  const [editStudent, setEditStudent] = useState<Student | null>(null)
   380→  const [deleteStudent, setDeleteStudent] = useState<Student | null>(null)
   381→  const [passwordStudent, setPasswordStudent] = useState<Student | null>(null)
   382→
   383→  const { data: students, loading, refresh } = useFetch<Student[]>(
   384→    () => api.adminStudents(search || undefined).then((res) => res.data || res.students || res),
   385→    [search]
   386→  )
   387→
   388→  const { data: courses } = useFetch<Course[]>(
   389→    () => api.adminCourses().then((res) => res.data || res.courses || res),
   390→    []
   391→  )
   392→
   393→  const courseList = courses || []
   394→
   395→  const handleAdd = async (e: React.FormEvent<HTMLFormElement>) => {
   396→    e.preventDefault()
   397→    const fd = new FormData(e.currentTarget)
   398→    try {
   399→      await api.adminCreateStudent({
   400→        name: fd.get('name'), email: fd.get('email'), mobile: fd.get('mobile'),
   401→        password: fd.get('password'), courseId: fd.get('courseId'),
   402→      })
   403→      toast.success('Student created')
   404→      setAddOpen(false)
   405→      refresh()
   406→    } catch (err: any) { toast.error(err.message) }
   407→  }
   408→
   409→  const handleEdit = async (e: React.FormEvent<HTMLFormElement>) => {
   410→    e.preventDefault()
   411→    if (!editStudent) return
   412→    const fd = new FormData(e.currentTarget)
   413→    try {
   414→      await api.adminUpdateStudent(editStudent.id, {
   415→        name: fd.get('name'), email: fd.get('email'), mobile: fd.get('mobile'),
   416→        courseId: fd.get('courseId'), status: fd.get('status'),
   417→      })
   418→      toast.success('Student updated')
   419→      setEditStudent(null)
   420→      refresh()
   421→    } catch (err: any) { toast.error(err.message) }
   422→  }
   423→
   424→  const handleDelete = async () => {
   425→    if (!deleteStudent) return
   426→    try {
   427→      await api.adminDeleteStudent(deleteStudent.id)
   428→      toast.success('Student deleted')
   429→      setDeleteStudent(null)
   430→      refresh()
   431→    } catch (err: any) { toast.error(err.message) }
   432→  }
   433→
   434→  const handleChangePassword = async (e: React.FormEvent<HTMLFormElement>) => {
   435→    e.preventDefault()
   436→    if (!passwordStudent) return
   437→    const fd = new FormData(e.currentTarget)
   438→    try {
   439→      await api.adminChangePassword(passwordStudent.id, fd.get('password') as string)
   440→      toast.success('Password changed')
   441→      setPasswordStudent(null)
   442→    } catch (err: any) { toast.error(err.message) }
   443→  }
   444→
   445→  const studentList = students || []
   446→
   447→  return (
   448→    <div className="space-y-4">
   449→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   450→        <div className="relative w-full sm:w-72">
   451→          <Search className="absolute left-2.5 top-2.5 size-4 text-muted-foreground" />
   452→          <Input placeholder="Search students..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-8" />
   453→        </div>
   454→        <Button onClick={() => setAddOpen(true)}><Plus className="size-4" /> Add Student</Button>
   455→      </div>
   456→
   457→      <Card>
   458→        <CardContent className="p-0">
   459→          {loading ? <div className="p-4"><TableSkeleton cols={6} /></div> : (
   460→            <Table>
   461→              <TableHeader>
   462→                <TableRow>
   463→                  <TableHead>Name</TableHead>
   464→                  <TableHead>Email</TableHead>
   465→                  <TableHead className="hidden md:table-cell">Mobile</TableHead>
   466→                  <TableHead>Course</TableHead>
   467→                  <TableHead>Status</TableHead>
   468→                  <TableHead className="text-right">Actions</TableHead>
   469→                </TableRow>
   470→              </TableHeader>
   471→              <TableBody>
   472→                {!studentList.length ? (
   473→                  <TableRow><TableCell colSpan={6} className="text-center py-8 text-muted-foreground">No students found</TableCell></TableRow>
   474→                ) : studentList.map((s) => (
   475→                  <TableRow key={s.id}>
   476→                    <TableCell className="font-medium">{s.name}</TableCell>
   477→                    <TableCell className="text-sm">{s.email}</TableCell>
   478→                    <TableCell className="hidden md:table-cell">{s.mobile}</TableCell>
   479→                    <TableCell>{s.course?.title || '—'}</TableCell>
   480→                    <TableCell>
   481→                      <Badge variant={s.status === 'APPROVED' ? 'default' : s.status === 'REJECTED' ? 'destructive' : 'secondary'}>
   482→                        {s.status}
   483→                      </Badge>
   484→                    </TableCell>
   485→                    <TableCell className="text-right">
   486→                      <div className="flex items-center justify-end gap-1">
   487→                        <Button variant="ghost" size="icon" onClick={() => setEditStudent(s)} title="Edit"><Pencil className="size-4" /></Button>
   488→                        <Button variant="ghost" size="icon" onClick={() => setPasswordStudent(s)} title="Change Password"><Lock className="size-4" /></Button>
   489→                        <Button variant="ghost" size="icon" onClick={() => setDeleteStudent(s)} title="Delete"><Trash2 className="size-4 text-destructive" /></Button>
   490→                      </div>
   491→                    </TableCell>
   492→                  </TableRow>
   493→                ))}
   494→              </TableBody>
   495→            </Table>
   496→          )}
   497→        </CardContent>
   498→      </Card>
   499→
   500→      {/* Add Student Dialog */}
   501→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   502→        <DialogContent>
   503→          <DialogHeader><DialogTitle>Add Student</DialogTitle><DialogDescription>Create a new student account</DialogDescription></DialogHeader>
   504→          <form onSubmit={handleAdd} className="space-y-3">
   505→            <div><Label>Name</Label><Input name="name" required /></div>
   506→            <div><Label>Email</Label><Input name="email" type="email" required /></div>
   507→            <div><Label>Mobile</Label><Input name="mobile" required /></div>
   508→            <div><Label>Password</Label><Input name="password" type="password" required /></div>
   509→            <div>
   510→              <Label>Course</Label>
   511→              <Select name="courseId">
   512→                <SelectTrigger className="w-full"><SelectValue placeholder="Select course" /></SelectTrigger>
   513→                <SelectContent>
   514→                  {courseList.map((c) => <SelectItem key={c.id} value={c.id}>{c.title}</SelectItem>)}
   515→                </SelectContent>
   516→              </Select>
   517→            </div>
   518→            <DialogFooter><Button type="submit">Create Student</Button></DialogFooter>
   519→          </form>
   520→        </DialogContent>
   521→      </Dialog>
   522→
   523→      {/* Edit Student Dialog */}
   524→      <Dialog open={!!editStudent} onOpenChange={(open) => !open && setEditStudent(null)}>
   525→        <DialogContent>
   526→          <DialogHeader><DialogTitle>Edit Student</DialogTitle></DialogHeader>
   527→          {editStudent && (
   528→            <form onSubmit={handleEdit} className="space-y-3">
   529→              <div><Label>Name</Label><Input name="name" defaultValue={editStudent.name} required /></div>
   530→              <div><Label>Email</Label><Input name="email" type="email" defaultValue={editStudent.email} required /></div>
   531→              <div><Label>Mobile</Label><Input name="mobile" defaultValue={editStudent.mobile} required /></div>
   532→              <div>

   535→                  <SelectTrigger className="w-full"><SelectValue placeholder="Select course" /></SelectTrigger>
   536→                  <SelectContent>
   537→                    {courseList.map((c) => <SelectItem key={c.id} value={c.id}>{c.title}</SelectItem>)}
   538→                  </SelectContent>
   539→                </Select>
   540→              </div>
   541→              <div>
   542→                <Label>Status</Label>
   543→                <Select name="status" defaultValue={editStudent.status}>
