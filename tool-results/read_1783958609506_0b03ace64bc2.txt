   187→function InlineLineChart({ data, height=200 }: { data:{day:string;signups:number}[]; height?:number }) {
   188→  if (!data.length) return null
   189→  const maxVal = Math.max(...data.map(d=>d.signups), 1)
   190→  const w = 500, h = height, padL = 40, padR = 20, padT = 20, padB = 40
   191→  const plotW = w - padL - padR, plotH = h - padT - padB
   192→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   193→  const xScale = (i:number) => padL + (i/(data.length-1))*plotW
   194→
   195→  const linePath = data.map((d,i)=>`${i===0?'M':'L'}${xScale(i)},${yScale(d.signups)}`).join(' ')
   196→  const areaPath = linePath + ` L${xScale(data.length-1)},${padT+plotH} L${padL},${padT+plotH} Z`
   197→  const yTicks = 5
   198→  const yStep = maxVal / yTicks
   199→
   200→  return (
   201→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   202→      {/* Grid lines */}
   203→      {Array.from({length:yTicks+1}).map((_,i)=>{
   204→        const val = Math.round(i*yStep)
   205→        const y = yScale(val)
   206→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   207→      })}
   208→      {/* Area fill */}
   209→      <path d={areaPath} fill="#10b981" fillOpacity={0.08} />
   210→      {/* Line */}
   211→      <path d={linePath} fill="none" stroke="#10b981" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
   212→      {/* Dots + labels */}
   213→      {data.map((d,i)=><g key={i}>
   214→        <circle cx={xScale(i)} cy={yScale(d.signups)} r="4" fill="#10b981" />
   215→        <text x={xScale(i)} y={h-8} textAnchor="middle" className="fill-muted-foreground" fontSize="11">{d.day}</text>
   216→      </g>)}
   217→    </svg>
   218→  )
   219→}

   221→function InlineBarChart({ data, height=220 }: { data:TopPerformer[]; height?:number }) {
   222→  if (!data.length) return null
   223→  const maxVal = Math.max(...data.map(d=>d.studyHours), 1)
   224→  const barW = Math.min(40, 400/data.length - 8)
   225→  const w = 500, h = height, padL = 40, padR = 20, padT = 10, padB = 60
   226→  const plotW = w - padL - padR, plotH = h - padT - padB
   227→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   228→  const gap = plotW / data.length
   229→
   230→  return (
   231→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>

   235→        const val = Math.round(maxVal*pct)

   254→function useFetch<T>(fetcher:()=>Promise<T>, deps:any[]) {
   255→  const [data, setData] = useState<T|null>(null)
   256→  const [loading, setLoading] = useState(true)
   257→  const [error, setError] = useState<string|null>(null)
   258→  const [refreshKey, setRefreshKey] = useState(0)
   259→  const refresh = useCallback(()=>setRefreshKey(k=>k+1), [])
   260→  useEffect(()=>{
   261→    let c=false
   262→    fetcher().then(r=>{if(c)return;setData(r);setError(null);setLoading(false)}).catch(e=>{if(c)return;setError(e.message||'Failed');setLoading(false)})
   263→    return()=>{c=true}
   264→  }, [...deps, refreshKey])
   265→  return { data, loading, error, refresh, setData }
   266→}

   268→function usePagination<T>(items:T[], pageSize=10) {
   269→  const [page, setPage] = useState(1)
   270→  const safeItems = Array.isArray(items) ? items : []
   271→  const totalPages = Math.max(1, Math.ceil(safeItems.length/pageSize))
   272→  const safePage = Math.min(page, totalPages)
   273→  const paginated = safeItems.slice((safePage-1)*pageSize, safePage*pageSize)
   274→  const goTo = (p:number)=>setPage(Math.max(1,Math.min(p,totalPages)))
   275→  if (page>totalPages&&totalPages>0) setPage(totalPages)
   276→  return { page:safePage, totalPages, paginated, goTo, setPage }
   277→}
