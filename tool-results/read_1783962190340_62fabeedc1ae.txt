    61→// Animated counter hook using IntersectionObserver (no framer-motion)
    62→function useAnimatedCounter(end: number, duration: number = 2000) {
    63→  const [count, setCount] = useState(0)
    64→  const [hasStarted, setHasStarted] = useState(false)
    65→  const ref = useRef<HTMLSpanElement>(null)
    66→
    67→  useEffect(() => {
    68→    const el = ref.current
    69→    if (!el) return
    70→    const observer = new IntersectionObserver(
    71→      ([entry]) => { if (entry.isIntersecting && !hasStarted) setHasStarted(true) },
    72→      { threshold: 0.3 }
    73→    )
    74→    observer.observe(el)
    75→    return () => observer.disconnect()
    76→  }, [hasStarted])
    77→
    78→  useEffect(() => {
    79→    if (!hasStarted) return
    80→    let startTime: number | null = null
    81→    const step = (timestamp: number) => {
    82→      if (!startTime) startTime = timestamp
    83→      const progress = Math.min((timestamp - startTime) / duration, 1)
    84→      setCount(Math.floor(progress * end))
    85→      if (progress < 1) requestAnimationFrame(step)
    86→      else setCount(end)
    87→    }
    88→    requestAnimationFrame(step)
    89→  }, [end, duration, hasStarted])
    90→
    91→  return { count, ref }
    92→}

    94→// Fade-in on scroll hook using IntersectionObserver
    95→function useFadeIn() {
    96→  const ref = useRef<HTMLDivElement>(null)
    97→  const [visible, setVisible] = useState(false)
    98→
    99→  useEffect(() => {
   100→    const el = ref.current
   101→    if (!el) return
   102→    const observer = new IntersectionObserver(
   103→      ([entry]) => { if (entry.isIntersecting) setVisible(true) },
   104→      { threshold: 0.1 }
   105→    )
   106→    observer.observe(el)
   107→    return () => observer.disconnect()
   108→  }, [])
   109→
   110→  return { ref, visible }
   111→}
