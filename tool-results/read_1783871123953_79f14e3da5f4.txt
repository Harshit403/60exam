   143→// ─── Helper: format time ────────────────────────────────────────────────
   144→
   145→function formatTimer(seconds: number): string {
   146→  const h = Math.floor(seconds / 3600)
   147→  const m = Math.floor((seconds % 3600) / 60)
   148→  const s = seconds % 60
   149→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   150→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   151→}
   152→
   153→function formatMinutes(min: number): string {
   154→  if (min < 60) return `${min}m`
   155→  const h = Math.floor(min / 60)
   156→  const m = min % 60
   157→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   158→}
   159→
   160→function formatDate(date: Date): string {
   161→  return date.toISOString().split('T')[0]
   162→}
