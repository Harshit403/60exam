    96→function InlineLineChart({ data, height=200 }: { data:{day:string;signups:number}[]; height?:number }) {
    97→  if (!data.length) return null
    98→  const maxVal = Math.max(...data.map(d=>d.signups), 1)
    99→  const w = 500, h = height, padL = 40, padR = 20, padT = 20, padB = 40
   100→  const plotW = w - padL - padR, plotH = h - padT - padB
   101→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   102→  const xScale = (i:number) => padL + (i/(data.length-1))*plotW
   103→
   104→  const linePath = data.map((d,i)=>`${i===0?'M':'L'}${xScale(i)},${yScale(d.signups)}`).join(' ')
   105→  const areaPath = linePath + ` L${xScale(data.length-1)},${padT+plotH} L${padL},${padT+plotH} Z`
   106→  const yTicks = 5
   107→  const yStep = maxVal / yTicks
   108→
   109→  return (
   110→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   111→      {/* Grid lines */}
   112→      {Array.from({length:yTicks+1}).map((_,i)=>{
   113→        const val = Math.round(i*yStep)
   114→        const y = yScale(val)
   115→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   116→      })}
   117→      {/* Area fill */}
   118→      <path d={areaPath} fill="#10b981" fillOpacity={0.08} />
   119→      {/* Line */}
   120→      <path d={linePath} fill="none" stroke="#10b981" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
   121→      {/* Dots + labels */}
   122→      {data.map((d,i)=><g key={i}>
   123→        <circle cx={xScale(i)} cy={yScale(d.signups)} r="4" fill="#10b981" />
   124→        <text x={xScale(i)} y={h-8} textAnchor="middle" className="fill-muted-foreground" fontSize="11">{d.day}</text>
   125→      </g>)}
   126→    </svg>
   127→  )
   128→}

   130→function InlineBarChart({ data, height=220 }: { data:TopPerformer[]; height?:number }) {
   131→  if (!data.length) return null
   132→  const maxVal = Math.max(...data.map(d=>d.studyHours), 1)
   133→  const barW = Math.min(40, 400/data.length - 8)
   134→  const w = 500, h = height, padL = 40, padR = 20, padT = 10, padB = 60
   135→  const plotW = w - padL - padR, plotH = h - padT - padB
   136→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   137→  const gap = plotW / data.length
   138→
   139→  return (
   140→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   141→      {/* Grid */}
   142→      {[0,0.25,0.5,0.75,1].map((pct,i)=>{
   143→        const y = yScale(maxVal*pct)
   144→        const val = Math.round(maxVal*pct)
   145→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   146→      })}
   147→      {/* Bars */}
   148→      {data.map((d,i)=>{
   149→        const x = padL + i*gap + gap/2 - barW/2
   150→        const barH = (d.studyHours/maxVal)*plotH
   151→        const y = padT + plotH - barH
   152→        const name = d.name.length > 8 ? d.name.slice(0,7)+'…' : d.name
   153→        return <g key={i}>
   154→          <rect x={x} y={y} width={barW} height={barH} fill="#f59e0b" rx="3" className="transition-opacity hover:opacity-80" />
   155→          <text x={padL+i*gap+gap/2} y={h-28} textAnchor="middle" className="fill-muted-foreground" fontSize="10" transform={`rotate(-25,${padL+i*gap+gap/2},${h-28})`}>{name}</text>
   156→        </g>
   157→      })}
   158→    </svg>
   159→  )
   160→}

   162→// ─── Hooks ───────────────────────────────────────────────────────
   163→function useFetch<T>(fetcher:()=>Promise<T>, deps:any[]) {
   164→  const [data, setData] = useState<T|null>(null)
   165→  const [loading, setLoading] = useState(true)
   166→  const [error, setError] = useState<string|null>(null)
   167→  const [refreshKey, setRefreshKey] = useState(0)
   168→  const refresh = useCallback(()=>setRefreshKey(k=>k+1), [])
   169→  useEffect(()=>{
   170→    let c=false
   171→    fetcher().then(r=>{if(c)return;setData(r);setError(null);setLoading(false)}).catch(e=>{if(c)return;setError(e.message||'Failed');setLoading(false)})
   172→    return()=>{c=true}
   173→  }, [...deps, refreshKey])
   174→  return { data, loading, error, refresh, setData }
   175→}

   177→function usePagination<T>(items:T[], pageSize=10) {
   178→  const [page, setPage] = useState(1)
   179→  const totalPages = Math.max(1, Math.ceil(items.length/pageSize))
   180→  const safePage = Math.min(page, totalPages)
   181→  const paginated = items.slice((safePage-1)*pageSize, safePage*pageSize)
   182→  const goTo = (p:number)=>setPage(Math.max(1,Math.min(p,totalPages)))
   183→  if (page>totalPages&&totalPages>0) setPage(totalPages)
   184→  return { page:safePage, totalPages, paginated, goTo, setPage }
   185→}

   187→function exportCSV(filename:string, headers:string[], rows:string[][]) {
   188→  const csv = [headers.join(','),...rows.map(r=>r.map(c=>`"${c.replace(/"/g,'""')}"`).join(','))].join('\n')
   189→  const blob = new Blob([csv],{type:'text/csv;charset=utf-8;'})
   190→  const url = URL.createObjectURL(blob)
   191→  const a = document.createElement('a'); a.href=url; a.download=`${filename}.csv`; a.click()
   192→  URL.revokeObjectURL(url)
   193→}

   840→function DiscussionsPage() {

   874→    </AlertDialogContent></AlertDialog>

   878→// ─── 10. Change Password Dialog ──────────────────────────────────
   879→function ChangePasswordDialog({ open, onOpenChange }: { open:boolean; onOpenChange:(open:boolean)=>void }) {
   880→  const [currentPassword,setCurrentPassword]=useState(''), [newPassword,setNewPassword]=useState(''), [confirmPassword,setConfirmPassword]=useState(''), [submitting,setSubmitting]=useState(false)
   881→
   882→  const handleSubmit=async(e:React.FormEvent)=>{
   883→    e.preventDefault()
   884→    if(newPassword!==confirmPassword){toast.error('New passwords do not match');return}
   885→    if(newPassword.length<6){toast.error('Password must be at least 6 characters');return}
   886→    setSubmitting(true)
   887→    try{await api.adminChangeOwnPassword(currentPassword,newPassword);toast.success('Password changed successfully');onOpenChange(false);setCurrentPassword('');setNewPassword('');setConfirmPassword('')}
   888→    catch(e:any){toast.error(e.message||'Failed to change password')}finally{setSubmitting(false)}
   889→  }
   890→
   891→  return <Dialog open={open} onOpenChange={onOpenChange}><DialogContent>
   892→    <DialogHeader><DialogTitle>Change Password</DialogTitle><DialogDescription>Update your admin account password</DialogDescription></DialogHeader>
   893→    <form onSubmit={handleSubmit} className="space-y-4">
   894→      <IconField icon={Lock} label="Current Password" type="password" value={currentPassword} onChange={e=>setCurrentPassword(e.target.value)} required />
   895→      <IconField icon={Lock} label="New Password" type="password" value={newPassword} onChange={e=>setNewPassword(e.target.value)} required minLength={6} />
   896→      <IconField icon={Lock} label="Confirm New Password" type="password" value={confirmPassword} onChange={e=>setConfirmPassword(e.target.value)} required minLength={6} />
   897→      <DialogFooter><Button type="submit" disabled={submitting}>{submitting&&<Loader2 className="size-4 animate-spin" />} Change Password</Button></DialogFooter>
   898→    </form>
   899→  </DialogContent></Dialog>
   900→}
