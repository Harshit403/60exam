   161→// ─── Helper: Mini Sparkline ───────────────────────────────────────
   162→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   163→  const max = Math.max(...data, 1)
   164→  const points = data.map((v, i) => {
   165→    const x = (i / (data.length - 1)) * 60
   166→    const y = 24 - (v / max) * 20
   167→    return `${x},${y}`
   168→  }).join(' ')
   169→
   170→  return (
   171→    <svg width="64" height="28" className="opacity-60">
   172→      <polyline fill="none" stroke={color} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" points={points} />
   173→    </svg>
   174→  )
   175→}

   197→// ─── Custom hook: data fetcher with refresh ──────────────────────
   198→function useFetch<T>(fetcher: () => Promise<T>, deps: any[]) {
   199→  const [data, setData] = useState<T | null>(null)
   200→  const [loading, setLoading] = useState(true)
   201→  const [error, setError] = useState<string | null>(null)
   202→  const [refreshKey, setRefreshKey] = useState(0)
   203→
   204→  const refresh = useCallback(() => setRefreshKey((k) => k + 1), [])
   205→
   206→  useEffect(() => {
   207→    let cancelled = false
   208→    fetcher()
   209→      .then((res) => {
   210→        if (cancelled) return
   211→        setData(res)
   212→        setError(null)
   213→        setLoading(false)
   214→      })
   215→      .catch((err) => {
   216→        if (cancelled) return
   217→        setError(err.message || 'Failed to fetch')
   218→        setLoading(false)
   219→      })
   220→    return () => { cancelled = true }
   221→  }, [...deps, refreshKey])
   222→
   223→  return { data, loading, error, refresh, setData }
   224→}

   226→// ─── Helper: Pagination ──────────────────────────────────────────
   227→function usePagination<T>(items: T[], pageSize = 10) {
   228→  const [page, setPage] = useState(1)
   229→  const totalPages = Math.max(1, Math.ceil(items.length / pageSize))
   230→  const safePage = Math.min(page, totalPages)
   231→  const paginated = items.slice((safePage - 1) * pageSize, safePage * pageSize)
   232→
   233→  const goTo = (p: number) => setPage(Math.max(1, Math.min(p, totalPages)))
   234→
   235→  // Reset page if items shrink and current page is out of range
   236→  if (page > totalPages && totalPages > 0) {
   237→    setPage(totalPages)
   238→  }
   239→
   240→  return { page: safePage, totalPages, paginated, goTo, setPage }
   241→}

   243→// ─── Helper: Export CSV ──────────────────────────────────────────
   244→function exportCSV(filename: string, headers: string[], rows: string[][]) {
   245→  const csv = [headers.join(','), ...rows.map((r) => r.map((c) => `"${c.replace(/"/g, '""')}"`).join(','))].join('\n')
   246→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   247→  const url = URL.createObjectURL(blob)
   248→  const a = document.createElement('a')
   249→  a.href = url
   250→  a.download = `${filename}.csv`
   251→  a.click()
   252→  URL.revokeObjectURL(url)
   253→}
