    28→// ─── Timer Persistence Helpers ─────────────────────────────────────────
    29→const TIMER_STORAGE_KEY = 'mission-cs-pomodoro-state'
    30→
    31→interface TimerPersistState {
    32→  selectedSubjectId: string
    33→  selectedChapterId: string
    34→  timerSeconds: number
    35→  timerTotalSeconds: number
    36→  timerRunning: boolean
    37→  timerPaused: boolean
    38→  timestamp: number // when the state was saved
    39→  sessionQuote: string
    40→}
    41→
    42→function saveTimerState(state: TimerPersistState) {
    43→  try { localStorage.setItem(TIMER_STORAGE_KEY, JSON.stringify(state)) } catch {}
    44→}
    45→
    46→function loadTimerState(): TimerPersistState | null {
    47→  try {
    48→    const raw = localStorage.getItem(TIMER_STORAGE_KEY)
    49→    if (!raw) return null
    50→    const state = JSON.parse(raw) as TimerPersistState
    51→    // If timer was running, calculate elapsed time since last save
    52→    if (state.timerRunning && !state.timerPaused) {
    53→      const elapsed = Math.floor((Date.now() - state.timestamp) / 1000)
    54→      state.timerSeconds = Math.max(0, state.timerSeconds - elapsed)
    55→      if (state.timerSeconds <= 0) {
    56→        state.timerRunning = false
    57→        state.timerPaused = false
    58→      }
    59→    }
    60→    return state
    61→  } catch { return null }
    62→}
    63→
    64→function clearTimerState() {
    65→  try { localStorage.removeItem(TIMER_STORAGE_KEY) } catch {}
    66→}
