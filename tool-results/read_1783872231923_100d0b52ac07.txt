   249→function formatTimer(seconds: number): string {
   250→  const h = Math.floor(seconds / 3600)
   251→  const m = Math.floor((seconds % 3600) / 60)
   252→  const s = seconds % 60
   253→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   254→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   255→}
   256→
   257→function formatMinutes(min: number): string {
   258→  if (min < 60) return `${min}m`
   259→  const h = Math.floor(min / 60)
   260→  const m = min % 60
   261→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   262→}
   263→
   264→function formatDate(date: Date): string {
   265→  return date.toISOString().split('T')[0]
   266→}

   270→function CircularProgressRing({
   271→  size = 120,
   272→  strokeWidth = 8,
   273→  progress = 0,
   274→  color = '#10b981',
   275→  trackColor = 'rgba(100,116,139,0.2)',
   276→  children,

   282→  color?: string
   283→  trackColor?: string
   284→  children?: React.ReactNode
   285→  className?: string
   286→}) {
   287→  const radius = (size - strokeWidth) / 2
   288→  const circumference = radius * 2 * Math.PI
   289→  const offset = circumference - (Math.min(progress, 100) / 100) * circumference

   325→function TimerCountdownRing({
   326→  size = 240,
   327→  strokeWidth = 10,
   328→  totalSeconds = 0,
   329→  remainingSeconds = 0,
   330→}: {
   331→  size?: number
   332→  strokeWidth?: number
   333→  totalSeconds?: number
   334→  remainingSeconds?: number
   335→}) {
   336→  const radius = (size - strokeWidth) / 2
   337→  const circumference = radius * 2 * Math.PI
   338→  const progress = totalSeconds > 0 ? remainingSeconds / totalSeconds : 0
   339→  const offset = circumference - progress * circumference
   340→  const center = size / 2
   341→
   342→  // Color transitions: green → yellow → red
   343→  let ringColor = '#10b981' // emerald
   344→  let glowColor = 'rgba(16,185,129,0.3)'
   345→  if (progress < 0.25) {
   346→    ringColor = '#ef4444' // red
   347→    glowColor = 'rgba(239,68,68,0.4)'
   348→  } else if (progress < 0.5) {
   349→    ringColor = '#f59e0b' // amber
   350→    glowColor = 'rgba(245,158,11,0.3)'
   351→  }
   352→
   353→  return (
   354→    <svg width={size} height={size} className="transform -rotate-90">
   355→      {/* Outer glow */}
   356→      <circle
   357→        cx={center}
   358→        cy={center}
   359→        r={radius + 4}
   360→        stroke={glowColor}
   361→        strokeWidth={strokeWidth + 8}
   362→        fill="none"
   363→        className="blur-sm"
   364→      />
   365→      {/* Track */}
   366→      <circle
   367→        cx={center}
   368→        cy={center}
   369→        r={radius}
   370→        stroke="rgba(100,116,139,0.15)"
   371→        strokeWidth={strokeWidth}
   372→        fill="none"
   373→      />
   374→      {/* Progress ring */}
   375→      <circle
   376→        cx={center}
   377→        cy={center}
   378→        r={radius}
   379→        stroke={ringColor}
   380→        strokeWidth={strokeWidth}
   381→        fill="none"
   382→        strokeDasharray={circumference}
   383→        strokeDashoffset={offset}
   384→        strokeLinecap="round"
   385→        style={{
   386→          transition: 'stroke-dashoffset 1s linear, stroke 2s ease',
   387→          filter: `drop-shadow(0 0 8px ${glowColor})`,
   388→        }}
   389→      />
   390→      {/* Tick marks */}
   391→      {Array.from({ length: 60 }).map((_, i) => {
   392→        const angle = (i / 60) * 360
   393→        const isMajor = i % 5 === 0
   394→        const innerR = radius - (isMajor ? 18 : 14)
   395→        const outerR = radius - 12
   396→        const x1 = center + innerR * Math.cos((angle * Math.PI) / 180)
   397→        const y1 = center + innerR * Math.sin((angle * Math.PI) / 180)
   398→        const x2 = center + outerR * Math.cos((angle * Math.PI) / 180)
   399→        const y2 = center + outerR * Math.sin((angle * Math.PI) / 180)
   400→        return (
   401→          <line
   402→            key={i}
   403→            x1={x1}
   404→            y1={y1}
   405→            x2={x2}
   406→            y2={y2}
   407→            stroke={isMajor ? 'rgba(100,116,139,0.3)' : 'rgba(100,116,139,0.15)'}
   408→            strokeWidth={isMajor ? 2 : 1}
   409→            transform={`rotate(90, ${center}, ${center})`}
   410→          />
   411→        )
   412→      })}
   413→    </svg>
   414→  )
   415→}

   419→function Confetti({ active }: { active: boolean }) {
   420→  const [pieces, setPieces] = useState<Array<{ id: number; x: number; color: string; delay: number; size: number }>>([])
   421→
   422→  useEffect(() => {
   423→    if (active) {
   424→      const newPieces = Array.from({ length: 40 }).map((_, i) => ({
   425→        id: i,
   426→        x: Math.random() * 100,
   427→        color: ['#10b981', '#f59e0b', '#ef4444', '#8b5cf6', '#3b82f6', '#ec4899', '#14b8a6'][Math.floor(Math.random() * 7)],
   428→        delay: Math.random() * 0.8,
   429→        size: Math.random() * 8 + 5,
   430→      }))
   431→      setPieces(newPieces)
   432→      const timeout = setTimeout(() => setPieces([]), 3000)
   433→      return () => clearTimeout(timeout)
   434→    } else {
   435→      setPieces([])
   436→    }
   437→  }, [active])
   438→
   439→  if (!active || pieces.length === 0) return null
   440→
   441→  return (
   442→    <div className="fixed inset-0 pointer-events-none z-[9999]">
   443→      {pieces.map((p) => (
   444→        <div
   445→          key={p.id}
   446→          className="confetti-piece"
   447→          style={{
   448→            left: `${p.x}%`,
   449→            backgroundColor: p.color,
   450→            width: `${p.size}px`,
   451→            height: `${p.size}px`,
   452→            borderRadius: Math.random() > 0.5 ? '50%' : '2px',
   453→            animationDelay: `${p.delay}s`,
   454→          }}
   455→        />
   456→      ))}
   457→    </div>
   458→  )
   459→}

   487→// ─── Study Heatmap Component ────────────────────────────────────────────
   488→
   489→function StudyHeatmap({ data }: { data: Record<string, number> }) {
   490→  const today = new Date()
   491→  const days = Array.from({ length: 30 }).map((_, i) => {
   492→    const d = new Date()
   493→    d.setDate(today.getDate() - (29 - i))
   494→    return formatDate(d)
   495→  })
   496→
   497→  const maxMin = Math.max(...Object.values(data), 1)
   498→
   499→  const getColor = (minutes: number) => {
   500→    if (minutes === 0) return 'bg-slate-100 dark:bg-slate-800'
   501→    const intensity = minutes / maxMin
   502→    if (intensity < 0.25) return 'bg-emerald-200 dark:bg-emerald-900'
   503→    if (intensity < 0.5) return 'bg-emerald-300 dark:bg-emerald-700'
   504→    if (intensity < 0.75) return 'bg-emerald-400 dark:bg-emerald-600'
   505→    return 'bg-emerald-500 dark:bg-emerald-500'
   506→  }
   507→
   508→  // Group into weeks (rows of 7 or fewer)
   509→  const weeks: string[][] = []
   510→  // First partial week
   511→  const firstDayOfWeek = new Date(days[0] + 'T00:00:00').getDay()
   512→  let currentWeek: string[] = []
   513→  // Pad first week
   514→  for (let i = 0; i < firstDayOfWeek; i++) currentWeek.push('')
   515→  for (const day of days) {
   516→    currentWeek.push(day)
   517→    if (currentWeek.length === 7) {
   518→      weeks.push(currentWeek)
   519→      currentWeek = []
   520→    }
   521→  }
   522→  if (currentWeek.length > 0) weeks.push(currentWeek)
   523→
   524→  return (
   525→    <div className="space-y-2">
   526→      <div className="flex gap-0.5 overflow-x-auto pb-2">
   527→        {/* Simple grid: 30 cells in 5 rows of 6 (cols) */}
   528→        <div className="flex flex-col gap-0.5">
   529→          {days.map((day, i) => {
   530→            const minutes = data[day] || 0
   531→            return (
   532→              <TooltipProvider key={i}>
   533→                <Tooltip>
   534→                  <TooltipTrigger asChild>
   535→                    <div
   536→                      className={`w-4 h-4 rounded-[2px] ${getColor(minutes)} cursor-pointer transition-colors hover:ring-1 hover:ring-slate-400`}
   537→                    />
   538→                  </TooltipTrigger>
   539→                  <TooltipContent side="top" className="text-xs">
   540→                    <p className="font-medium">{new Date(day + 'T00:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric' })}</p>
   541→                    <p className="text-slate-400">{minutes > 0 ? formatMinutes(minutes) : 'No study'}</p>
   542→                  </TooltipContent>
   543→                </Tooltip>
   544→              </TooltipProvider>
   545→            )
   546→          })}
   547→        </div>
   548→      </div>
   549→      {/* Legend */}
   550→      <div className="flex items-center gap-1 text-[10px] text-slate-400">
   551→        <span>Less</span>
   552→        <div className="w-3 h-3 rounded-[2px] bg-slate-100 dark:bg-slate-800" />
   553→        <div className="w-3 h-3 rounded-[2px] bg-emerald-200 dark:bg-emerald-900" />
   554→        <div className="w-3 h-3 rounded-[2px] bg-emerald-300 dark:bg-emerald-700" />
   555→        <div className="w-3 h-3 rounded-[2px] bg-emerald-400 dark:bg-emerald-600" />
   556→        <div className="w-3 h-3 rounded-[2px] bg-emerald-500 dark:bg-emerald-500" />
   557→        <span>More</span>
   558→      </div>
   559→    </div>
   560→  )
   561→}
