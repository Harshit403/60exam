   181→function InlineLineChart({ data, height=200 }: { data:{day:string;signups:number}[]; height?:number }) {
   182→  if (!data.length) return null
   183→  const maxVal = Math.max(...data.map(d=>d.signups), 1)
   184→  const w = 500, h = height, padL = 40, padR = 20, padT = 20, padB = 40
   185→  const plotW = w - padL - padR, plotH = h - padT - padB
   186→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   187→  const xScale = (i:number) => padL + (i/(data.length-1))*plotW
   188→
   189→  const linePath = data.map((d,i)=>`${i===0?'M':'L'}${xScale(i)},${yScale(d.signups)}`).join(' ')
   190→  const areaPath = linePath + ` L${xScale(data.length-1)},${padT+plotH} L${padL},${padT+plotH} Z`
   191→  const yTicks = 5
   192→  const yStep = maxVal / yTicks
   193→
   194→  return (
   195→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   196→      {/* Grid lines */}
   197→      {Array.from({length:yTicks+1}).map((_,i)=>{
   198→        const val = Math.round(i*yStep)
   199→        const y = yScale(val)
   200→        return <g key={i}><line x1={padL} y1={y} x2={w-padR} y2={y} stroke="currentColor" strokeOpacity={0.08} strokeDasharray="4 4" /><text x={padL-8} y={y+4} textAnchor="end" className="fill-muted-foreground" fontSize="11">{val}</text></g>
   201→      })}
   202→      {/* Area fill */}
   203→      <path d={areaPath} fill="#10b981" fillOpacity={0.08} />
   204→      {/* Line */}
   205→      <path d={linePath} fill="none" stroke="#10b981" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
   206→      {/* Dots + labels */}
   207→      {data.map((d,i)=><g key={i}>
   208→        <circle cx={xScale(i)} cy={yScale(d.signups)} r="4" fill="#10b981" />
   209→        <text x={xScale(i)} y={h-8} textAnchor="middle" className="fill-muted-foreground" fontSize="11">{d.day}</text>
   210→      </g>)}
   211→    </svg>
   212→  )
   213→}

   215→function InlineBarChart({ data, height=220 }: { data:TopPerformer[]; height?:number }) {
   216→  if (!data.length) return null
   217→  const maxVal = Math.max(...data.map(d=>d.studyHours), 1)
   218→  const barW = Math.min(40, 400/data.length - 8)
   219→  const w = 500, h = height, padL = 40, padR = 20, padT = 10, padB = 60
   220→  const plotW = w - padL - padR, plotH = h - padT - padB
   221→  const yScale = (v:number) => padT + plotH - (v/maxVal)*plotH
   222→  const gap = plotW / data.length
   223→
   224→  return (
   225→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{maxHeight:height}}>
   226→      {/* Grid */}
   227→      {[0,0.25,0.5,0.75,1].map((pct,i)=>{
   228→        const y = yScale(maxVal*pct)
   229→        const val = Math.round(maxVal*pct)

   247→// ─── Hooks ───────────────────────────────────────────────────────
   248→function useFetch<T>(fetcher:()=>Promise<T>, deps:any[]) {
   249→  const [data, setData] = useState<T|null>(null)
   250→  const [loading, setLoading] = useState(true)
   251→  const [error, setError] = useState<string|null>(null)
   252→  const [refreshKey, setRefreshKey] = useState(0)
   253→  const refresh = useCallback(()=>setRefreshKey(k=>k+1), [])
   254→  useEffect(()=>{
   255→    let c=false
   256→    fetcher().then(r=>{if(c)return;setData(r);setError(null);setLoading(false)}).catch(e=>{if(c)return;setError(e.message||'Failed');setLoading(false)})
   257→    return()=>{c=true}
   258→  }, [...deps, refreshKey])
   259→  return { data, loading, error, refresh, setData }
   260→}

   262→function usePagination<T>(items:T[], pageSize=10) {
   263→  const [page, setPage] = useState(1)
   264→  const safeItems = Array.isArray(items) ? items : []
   265→  const totalPages = Math.max(1, Math.ceil(safeItems.length/pageSize))
   266→  const safePage = Math.min(page, totalPages)
   267→  const paginated = safeItems.slice((safePage-1)*pageSize, safePage*pageSize)
   268→  const goTo = (p:number)=>setPage(Math.max(1,Math.min(p,totalPages)))
   269→  if (page>totalPages&&totalPages>0) setPage(totalPages)
   270→  return { page:safePage, totalPages, paginated, goTo, setPage }
   271→}
