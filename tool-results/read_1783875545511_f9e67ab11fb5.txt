   189→function formatTimer(seconds: number): string {
   190→  const h = Math.floor(seconds / 3600)
   191→  const m = Math.floor((seconds % 3600) / 60)
   192→  const s = seconds % 60
   193→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   194→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   195→}
   196→
   197→function formatMinutes(min: number): string {
   198→  if (min < 60) return `${min}m`
   199→  const h = Math.floor(min / 60)
   200→  const m = min % 60
   201→  return m > 0 ? `${h}h ${m}m` : `${h}h`
   202→}
   203→
   204→function formatDate(date: Date): string {
   205→  return date.toISOString().split('T')[0]
   206→}

   208→// ─── Circular Progress Ring ─────────────────────────────────────────────
   209→
   210→function CircularProgressRing({
   211→  size = 120, strokeWidth = 8, progress = 0, color = '#10b981',
   212→  trackColor = 'rgba(100,116,139,0.2)', children, className = '',
   213→}: {
   214→  size?: number; strokeWidth?: number; progress?: number; color?: string
   215→  trackColor?: string; children?: React.ReactNode; className?: string
   216→}) {
   217→  const radius = (size - strokeWidth) / 2
   218→  const circumference = radius * 2 * Math.PI
   219→  const offset = circumference - (Math.min(progress, 100) / 100) * circumference
   220→  const center = size / 2
   221→
   222→  return (
   223→    <div className={`relative inline-flex items-center justify-center ${className}`}>
   224→      <svg width={size} height={size} className="transform -rotate-90">
   225→        <circle cx={center} cy={center} r={radius} stroke={trackColor} strokeWidth={strokeWidth} fill="none" />
   226→        <circle cx={center} cy={center} r={radius} stroke={color} strokeWidth={strokeWidth} fill="none"
   227→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
   228→          style={{ transition: 'stroke-dashoffset 0.5s ease' }} />
   229→      </svg>
   230→      <div className="absolute inset-0 flex items-center justify-center">{children}</div>
   231→    </div>
   232→  )
   233→}

   235→// ─── Timer Countdown Ring ───────────────────────────────────────────────
   236→
   237→function TimerCountdownRing({
   238→  size = 240, strokeWidth = 10, totalSeconds = 0, remainingSeconds = 0,
   239→}: {
   240→  size?: number; strokeWidth?: number; totalSeconds?: number; remainingSeconds?: number
   241→}) {
   242→  const radius = (size - strokeWidth) / 2
   243→  const circumference = radius * 2 * Math.PI
   244→  const progress = totalSeconds > 0 ? remainingSeconds / totalSeconds : 0
   245→  const offset = circumference - progress * circumference
   246→  const center = size / 2
   247→
   248→  let ringColor = '#10b981', glowColor = 'rgba(16,185,129,0.3)'
   249→  if (progress < 0.25) { ringColor = '#ef4444'; glowColor = 'rgba(239,68,68,0.4)' }
   250→  else if (progress < 0.5) { ringColor = '#f59e0b'; glowColor = 'rgba(245,158,11,0.3)' }
   251→
   252→  return (
   253→    <svg width={size} height={size} className="transform -rotate-90">
   254→      <circle cx={center} cy={center} r={radius + 4} stroke={glowColor}
   255→        strokeWidth={strokeWidth + 8} fill="none" className="blur-sm" />
   256→      <circle cx={center} cy={center} r={radius} stroke="rgba(100,116,139,0.15)"
   257→        strokeWidth={strokeWidth} fill="none" />
   258→      <circle cx={center} cy={center} r={radius} stroke={ringColor} strokeWidth={strokeWidth}
   259→        fill="none" strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
   260→        style={{ transition: 'stroke-dashoffset 1s linear, stroke 2s ease', filter: `drop-shadow(0 0 8px ${glowColor})` }} />
   261→      {Array.from({ length: 60 }).map((_, i) => {
   262→        const angle = (i / 60) * 360
   263→        const isMajor = i % 5 === 0
   264→        const innerR = radius - (isMajor ? 18 : 14)
   265→        const outerR = radius - 12
   266→        const rad = (angle * Math.PI) / 180
   267→        return (
   268→          <line key={i}
   269→            x1={center + innerR * Math.cos(rad)} y1={center + innerR * Math.sin(rad)}
   270→            x2={center + outerR * Math.cos(rad)} y2={center + outerR * Math.sin(rad)}
   271→            stroke={isMajor ? 'rgba(100,116,139,0.3)' : 'rgba(100,116,139,0.15)'}
   272→            strokeWidth={isMajor ? 2 : 1} transform={`rotate(90, ${center}, ${center})`} />
   273→        )
   274→      })}
   275→    </svg>
   276→  )
   277→}

   281→function Confetti({ active }: { active: boolean }) {
   282→  const pieces = useMemo(() => {
   283→    if (!active) return []
   284→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#8b5cf6', '#ec4899', '#14b8a6']
   285→    return Array.from({ length: 30 }).map((_, i) => ({
   286→      id: i, x: Math.random() * 100,
   287→      color: colors[Math.floor(Math.random() * colors.length)],
   288→      delay: Math.random() * 0.8, size: Math.random() * 8 + 5,
   289→    }))
   290→  }, [active])
   291→
   292→  const [visible, setVisible] = useState(false)
   293→
   294→  useEffect(() => {
   295→    if (active) {
   296→      setVisible(true)
   297→      const timeout = setTimeout(() => setVisible(false), 3000)
   298→      return () => clearTimeout(timeout)
   299→    } else { setVisible(false) }
   300→  }, [active])
   301→
   302→  if (!visible || pieces.length === 0) return null
   303→
   304→  return (
   305→    <div className="fixed inset-0 pointer-events-none z-[9999]">
   306→      {pieces.map(p => (
   307→        <div key={p.id} className="confetti-piece"
   308→          style={{ left: `${p.x}%`, backgroundColor: p.color, width: `${p.size}px`,
   309→            height: `${p.size}px`, borderRadius: p.id % 2 ? '50%' : '2px', animationDelay: `${p.delay}s` }} />
   310→      ))}
   311→    </div>
   312→  )
   313→}

   315→// ─── Study Heatmap ──────────────────────────────────────────────────────
   316→
   317→function StudyHeatmap({ data }: { data: Record<string, number> }) {
   318→  const today = new Date()
   319→  const days = Array.from({ length: 30 }).map((_, i) => {
   320→    const d = new Date(); d.setDate(today.getDate() - (29 - i)); return formatDate(d)
   321→  })
   322→  const maxMin = Math.max(...Object.values(data), 1)
   323→
   324→  const getColor = (minutes: number) => {
   325→    if (minutes === 0) return 'bg-slate-100 dark:bg-slate-800'
   326→    const intensity = minutes / maxMin
   327→    if (intensity < 0.25) return 'bg-emerald-200 dark:bg-emerald-900'
   328→    if (intensity < 0.5) return 'bg-emerald-300 dark:bg-emerald-700'
   329→    if (intensity < 0.75) return 'bg-emerald-400 dark:bg-emerald-600'
   330→    return 'bg-emerald-500 dark:bg-emerald-500'
   331→  }
   332→
   333→  return (
   334→    <div className="space-y-2">
   335→      <div className="flex gap-0.5 overflow-x-auto pb-2">
   336→        <div className="flex flex-col gap-0.5">
   337→          {days.map((day, i) => {
   338→            const minutes = data[day] || 0
   339→            return (
   340→              <TooltipProvider key={i}>
   341→                <Tooltip>
   342→                  <TooltipTrigger asChild>
   343→                    <div className={`w-4 h-4 rounded-[2px] ${getColor(minutes)} cursor-pointer transition-colors hover:ring-1 hover:ring-slate-400`} />
   344→                  </TooltipTrigger>
   345→                  <TooltipContent side="top" className="text-xs">
   346→                    <p className="font-medium">{new Date(day + 'T00:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric' })}</p>
   347→                    <p className="text-slate-400">{minutes > 0 ? formatMinutes(minutes) : 'No study'}</p>
   348→                  </TooltipContent>
   349→                </Tooltip>
   350→              </TooltipProvider>
   351→            )
   352→          })}
   353→        </div>
   354→      </div>
   355→      <div className="flex items-center gap-1 text-[10px] text-slate-400">
   356→        <span>Less</span>
   357→        <div className="w-3 h-3 rounded-[2px] bg-slate-100 dark:bg-slate-800" />
   358→        <div className="w-3 h-3 rounded-[2px] bg-emerald-200 dark:bg-emerald-900" />
   359→        <div className="w-3 h-3 rounded-[2px] bg-emerald-300 dark:bg-emerald-700" />
   360→        <div className="w-3 h-3 rounded-[2px] bg-emerald-400 dark:bg-emerald-600" />
   361→        <div className="w-3 h-3 rounded-[2px] bg-emerald-500 dark:bg-emerald-500" />
   362→        <span>More</span>
   363→      </div>
   364→    </div>
   365→  )
   366→}
