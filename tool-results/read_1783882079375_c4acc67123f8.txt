    60→// Animated counter hook using IntersectionObserver (no framer-motion)
    61→function useAnimatedCounter(end: number, duration: number = 2000) {
    62→  const [count, setCount] = useState(0)
    63→  const [hasStarted, setHasStarted] = useState(false)
    64→  const ref = useRef<HTMLSpanElement>(null)
    65→
    66→  useEffect(() => {
    67→    const el = ref.current
    68→    if (!el) return
    69→    const observer = new IntersectionObserver(
    70→      ([entry]) => { if (entry.isIntersecting && !hasStarted) setHasStarted(true) },
    71→      { threshold: 0.3 }
    72→    )
    73→    observer.observe(el)
    74→    return () => observer.disconnect()
    75→  }, [hasStarted])
    76→
    77→  useEffect(() => {
    78→    if (!hasStarted) return
    79→    let startTime: number | null = null
    80→    const step = (timestamp: number) => {
    81→      if (!startTime) startTime = timestamp
    82→      const progress = Math.min((timestamp - startTime) / duration, 1)
    83→      setCount(Math.floor(progress * end))
    84→      if (progress < 1) requestAnimationFrame(step)
    85→      else setCount(end)
    86→    }
    87→    requestAnimationFrame(step)
    88→  }, [end, duration, hasStarted])
    89→
    90→  return { count, ref }
    91→}

    93→// Fade-in on scroll hook using IntersectionObserver
    94→function useFadeIn() {
    95→  const ref = useRef<HTMLDivElement>(null)
    96→  const [visible, setVisible] = useState(false)
    97→
    98→  useEffect(() => {
    99→    const el = ref.current
   100→    if (!el) return
   101→    const observer = new IntersectionObserver(
   102→      ([entry]) => { if (entry.isIntersecting) setVisible(true) },
   103→      { threshold: 0.1 }
   104→    )
   105→    observer.observe(el)
   106→    return () => observer.disconnect()
   107→  }, [])
   108→
   109→  return { ref, visible }
   110→}
